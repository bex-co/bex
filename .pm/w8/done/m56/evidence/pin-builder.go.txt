package main

import (
 "encoding/json"
 "fmt"
 "os"
 "github.com/render-oss/cli/pkg/client"
 servicepkg "github.com/render-oss/cli/pkg/service"
 servicetypes "github.com/render-oss/cli/pkg/types/service"
)

func main() {
 raw,err:=os.ReadFile("source-service.json");must(err)
 var before client.Service;must(json.Unmarshal(raw,&before))
 empty:=""
 args:=[]string{"--pre-deploy-command","","--health-check-path",""}
 input:=servicetypes.ServiceUpdateInput{ServiceIDOrName:before.Id,PreDeployCommand:&empty,HealthCheckPath:&empty}
 normalized,err:=servicetypes.NormalizeAndValidateUpdateInput(input);must(err)
 must(normalized.ValidateForServiceType(servicetypes.ServiceType(before.Type)))
 body,err:=servicepkg.BuildUpdateRequest(before,normalized);must(err)
 out:=map[string]any{"operation":"offline actual pinned normalizer, validators and update builder","network":false,"nativeWireIntercepted":false,"inputSource":"typed equivalent of observed flags; command parser not executed","flagArgs":args,"methodFromSource":"PATCH","pathFromSource":"/v1/services/"+before.Id,"body":body}
 result,err:=json.MarshalIndent(out,"","  ");must(err);fmt.Println(string(result))
}
func must(err error) { if err!=nil { panic(err) } }
