package apps

import "testing"

// TestQANumericEnvLiteralAccepted probes the production failing typed boundary,
// after the real pinned schema compiler has accepted each numeric literal.
func TestQANumericEnvLiteralAccepted(t *testing.T) {
 cases := []struct{name string; manifest string}{
{name: "envvalue-integer", manifest: `services:
  - type: web
    name: qa-20261008-e37b64-bp
    runtime: go
    repo: https://github.com/bex-co/bex
    rootDir: examples/hello-go
    plan: free
    buildCommand: go build -o app .
    startCommand: ./app
    envVars:
      - key: QA_MARKER
        value: 47
`},
{name: "root-group-numeric", manifest: `services:
- type: web
  name: qa-20261008-e37b64-bp
  runtime: go
  repo: https://github.com/bex-co/bex
  rootDir: examples/hello-go
  plan: free
  buildCommand: go build -o app .
  startCommand: ./app
  envVars:
  - key: QA_MARKER
    value: '47'
envVarGroups:
- name: qa-e37b64-eg
  envVars:
  - key: ANSWER
    value: 47
`},
{name: "ungrouped-service-numeric", manifest: `ungrouped:
  services:
  - type: web
    name: qa-20261008-e37b64-bp
    runtime: go
    repo: https://github.com/bex-co/bex
    rootDir: examples/hello-go
    plan: free
    buildCommand: go build -o app .
    startCommand: ./app
    envVars:
    - key: ANSWER
      value: 47
`},
{name: "project-service-numeric", manifest: `projects:
- name: qa-e37b64-project
  environments:
  - name: qa-e37b64-environment
    services:
    - type: web
      name: qa-20261008-e37b64-bp
      runtime: go
      repo: https://github.com/bex-co/bex
      rootDir: examples/hello-go
      plan: free
      buildCommand: go build -o app .
      startCommand: ./app
      envVars:
      - key: ANSWER
        value: 47
`},
{name: "project-group-numeric", manifest: `projects:
- name: qa-e37b64-project
  environments:
  - name: qa-e37b64-environment
    services:
    - type: web
      name: qa-20261008-e37b64-bp
      runtime: go
      repo: https://github.com/bex-co/bex
      rootDir: examples/hello-go
      plan: free
      buildCommand: go build -o app .
      startCommand: ./app
      envVars:
      - key: QA_MARKER
        value: '47'
    envVarGroups:
    - name: qa-e37b64-eg
      envVars:
      - key: ANSWER
        value: 47
`},
 }
 for _, c := range cases { t.Run(c.name, func(t *testing.T) {
  compiled, ir, problems := CompileBlueprintIR(c.manifest)
  if len(problems) != 0 {t.Fatalf("schema unexpectedly refused allowed numeric literal: %+v", problems)}
  if _, err := parseCompiledStack(blueprintParseOverrides{}, compiled, ir); err != nil {t.Fatalf("schema-valid numeric env value cannot reach typed stack: %v", err)}
 }) }
}
