import importlib.util,json,os,platform,subprocess,sys,time
from pathlib import Path

def tool(name):
    started=time.monotonic()
    try:
        r=subprocess.run([name,'--version'],text=True,capture_output=True,timeout=60)
        return {'status':r.returncode,'stdout':r.stdout,'stderr':r.stderr,'error':None,'elapsed_ms':round((time.monotonic()-started)*1000)}
    except FileNotFoundError:
        return {'status':None,'stdout':'','stderr':'','error':'ENOENT','elapsed_ms':round((time.monotonic()-started)*1000)}
    except subprocess.TimeoutExpired:
        return {'status':None,'stdout':'','stderr':'','error':'ETIMEDOUT','elapsed_ms':round((time.monotonic()-started)*1000)}
