const fs=require('node:fs'),http2=require('node:http2'),crypto=require('node:crypto');
const [url,file,stepText,receipt]=process.argv.slice(2),u=new URL(url),payload=fs.readFileSync(file),step=Number(stepText),start=process.hrtime.bigint();
const elapsed=()=>Number(process.hrtime.bigint()-start)/1e9;let sent=0,done=false,timer,deadline;
const out={client:process.version,protocol:'HTTP/2',url,bytes:payload.length,sha256:crypto.createHash('sha256').update(payload).digest('hex'),chunk_bytes:4096,interval_ms:step,writes:[],errors:[]};
const save=()=>{const t=receipt+'.tmp';fs.writeFileSync(t,JSON.stringify(out,null,2),{mode:0o600});fs.renameSync(t,receipt)};
const local=u.hostname==='127.0.0.1';const connection=http2.connect(u.origin,local?{rejectUnauthorized:false}:{});let request;
function finish(){if(done)return;done=true;clearTimeout(timer);clearTimeout(deadline);out.finished_at=elapsed();out.bytes_submitted=sent;out.complete=true;save();if(request&&!request.closed)request.close();connection.close();setTimeout(()=>connection.destroy(),100).unref();console.log(JSON.stringify({HTTP:out.response?.status,ALPN:out.ALPN,elapsed:out.finished_at,submitted:sent,response_complete:out.response?.complete,error:out.errors}));}
connection.on('error',e=>{out.errors.push({phase:'connection',code:e.code,message:e.message});finish()});connection.on('connect',()=>{out.ALPN=connection.alpnProtocol;save()});connection.on('goaway',(code,lastStreamID)=>{out.goaway={code,lastStreamID,at:elapsed()};save()});
request=connection.request({':method':'PUT',':path':u.pathname+u.search,'content-length':payload.length,'content-type':'application/octet-stream','user-agent':'qa442-owned-file-transfer'});
request.on('response',headers=>{out.response={status:headers[':status'],headers,started_at:elapsed(),body:''};save()});request.setEncoding('utf8');request.on('data',s=>{if(out.response)out.response.body+=s;save()});request.on('end',()=>{if(out.response){out.response.complete=true;try{out.response.decoded=JSON.parse(out.response.body)}catch{}}finish()});request.on('error',e=>{out.errors.push({phase:'request',code:e.code,message:e.message});finish()});request.on('close',()=>{if(!done){out.stream_closed_without_response_end=true;finish()}});
deadline=setTimeout(()=>{out.client_deadline_reached=true;finish()},120000);
function send(){if(done)return;if(sent===payload.length){out.upload_end_at=elapsed();save();request.end();return}const end=Math.min(sent+4096,payload.length),ok=request.write(payload.subarray(sent,end));sent=end;out.writes.push({at:elapsed(),bytes:sent,backpressure:!ok});save();if(ok)timer=setTimeout(send,step);else request.once('drain',()=>{timer=setTimeout(send,step)})}save();send();
