package build
import("encoding/base64";"encoding/json";"os";"path/filepath";"strings";"testing")
func TestQA358StaticDependencyObservation(t *testing.T){
 out:=os.Getenv("QA358_OUTPUT");source:=os.Getenv("QA358_SOURCE");if out==""||source==""{t.Fatal("private paths required")}
 for _,tc:=range []struct{name,skip string;nested,manifest bool}{{"root-default","",false,true},{"root-false","false",false,true},{"root-true","true",false,true},{"nested-default","",true,true},{"no-manifest","",false,false}}{t.Run(tc.name,func(t *testing.T){
  d:=t.TempDir();src:=filepath.Join(d,"source");selected:=src;if tc.nested{selected=filepath.Join(src,"site")};must(t,os.MkdirAll(selected,0700));if tc.manifest{for _,n:=range []string{"package.json","package-lock.json"}{b,e:=os.ReadFile(filepath.Join(source,n));if e!=nil{t.Fatal(e)};must(t,os.WriteFile(filepath.Join(selected,n),b,0600))}}
  env:="NODE_VERSION="+base64.StdEncoding.EncodeToString([]byte("24"))+"\n";if tc.skip!=""{env+="SKIP_INSTALL_DEPS="+base64.StdEncoding.EncodeToString([]byte(tc.skip))+"\n"};bundle:=filepath.Join(d,"env");must(t,os.WriteFile(bundle,[]byte(env),0600));o:=nativeOptions();o.BuildEnv=nil;o.Runtime="";o.StaticSite=true;o.StartCommand="";o.BuildCommand="npm run build-nocolor";if tc.nested{o.RootDir="site"};df:=filepath.Join(d,"Dockerfile");must(t,os.WriteFile(df,[]byte(nativeDockerfile(o)),0600));choices,e:=ResolveNativeBuild("node",bundle,df,selected);if e!=nil{t.Fatal(e)};resolved,e:=os.ReadFile(df);if e!=nil{t.Fatal(e)};if len(choices)!=1||choices[0].Line!="24"||strings.Contains(string(resolved),"\nCMD "){t.Fatalf("static/version control failed %+v",choices)}
  dest:=filepath.Join(out,tc.name);must(t,os.MkdirAll(dest,0700));must(t,os.WriteFile(filepath.Join(dest,"Dockerfile"),resolved,0600));record:=map[string]any{"case":tc.name,"choices":choices,"rootDir":o.RootDir,"manifest_and_npm_lock_at_selected_root":tc.manifest,"SKIP_INSTALL_DEPS":tc.skip,"actual_authored_build_command_preserved":strings.Contains(string(resolved),o.BuildCommand),"qualification":"real unchanged static generator/resolver; full generated application RUN retained for inspection; no Docker/install execution, cache bug or fixed regression claim"};b,e:=json.MarshalIndent(record,"","  ");if e!=nil{t.Fatal(e)};must(t,os.WriteFile(filepath.Join(dest,"observation.json"),b,0600));t.Logf("%s",b)
 })}
}
