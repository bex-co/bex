package apps

import (
	"context"
	"encoding/json"
	"github.com/bex-co/bex/lego/backend/internal/core"
	ids "github.com/bex-co/bex/lego/backend/internal/id"
	"github.com/bex-co/bex/lego/backend/internal/store"
	appv1alpha1 "github.com/bex-co/bex/lego/types/v1alpha1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"sigs.k8s.io/controller-runtime/pkg/client"
	"testing"
)

func TestQA230BlueprintOwnedIntentPersistence(t *testing.T) {
	for _, field := range []string{"source", "tier", "replicas", "idleTTL"} {
		t.Run(field, func(t *testing.T) {
			sid := ids.New(ids.Service)
			existing := &appv1alpha1.App{ObjectMeta: metav1.ObjectMeta{Name: "own-source", Namespace: "default", Generation: 7, Labels: map[string]string{store.LabelManagedBy: store.ManagedByValue, core.LabelAppID: sid, core.LabelTenant: "own-workspace"}}, Spec: appv1alpha1.AppSpec{Type: appv1alpha1.TypeWebService, Repo: "https://github.com/bex-co/bex", Branch: "main", Runtime: "node", Tier: "free", Port: 3000, Replicas: 1, IdleTTLSeconds: 300}}
			cl := fakeClient(existing)
			rec := &recordingStore{}
			svc := &Service{Base: &core.Base{Client: cl, Namespace: "default"}, Store: rec}
			final := existing.DeepCopy()
			switch field {
			case "source":
				final.Spec.Repo = "https://github.com/expressjs/express"
				final.Spec.Branch = "master"
			case "tier":
				final.Spec.Tier = "starter"
			case "replicas":
				final.Spec.Replicas = 2
			case "idleTTL":
				final.Spec.IdleTTLSeconds = 600
			}
			_, err := svc.patchChangedStackService(context.Background(), CreateRequest{Name: "own-source"}, existing, final, stackChanges{specChanged: true})
			if err != nil {
				t.Fatal(err)
			}
			stored := &appv1alpha1.App{}
			if err := cl.Get(context.Background(), client.ObjectKeyFromObject(existing), stored); err != nil {
				t.Fatal(err)
			}
			writes := rec.sourceCalls
			switch field {
			case "tier":
				writes = len(rec.tierCalls)
			case "replicas":
				writes = len(rec.replicasCalls)
			case "idleTTL":
				writes = len(rec.idleTTLCalls)
			}
			out := map[string]any{"field": field, "CR_patch_reaches_desired": stored.Spec.Repo == final.Spec.Repo && stored.Spec.Branch == final.Spec.Branch && stored.Spec.Tier == final.Spec.Tier && stored.Spec.Replicas == final.Spec.Replicas && stored.Spec.IdleTTLSeconds == final.Spec.IdleTTLSeconds, "canonical_intent_writes": writes, "deploy_rows_opened": len(rec.deployCalls), "managed_id": sid, "qualification": "actual patchChangedStackService with fake Kubernetes and recording IntentStore; no paid resources, live sibling changes or fixed regression claim"}
			b, _ := json.Marshal(out)
			t.Log("QA230_STORE " + string(b))
		})
	}
}
