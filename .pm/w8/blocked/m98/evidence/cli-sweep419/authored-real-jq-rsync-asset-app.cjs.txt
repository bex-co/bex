const fs=require('node:fs'),path=require('node:path'),os=require('node:os'),cp=require('node:child_process'),crypto=require('node:crypto'),http=require('node:http'),assert=require('node:assert/strict');
const ref=JSON.parse(fs.readFileSync('qa-assets-reference.json','utf8')),marker=JSON.parse(fs.readFileSync('qa-assets-marker.json','utf8')),phase=process.argv[2],sha=b=>crypto.createHash('sha256').update(b).digest('hex'),desc=b=>({bytes:b.length,sha256:sha(b)}),programSHA=sha(fs.readFileSync(__filename));
const filter='{release:.release,paths:[.assets[]|select(.publish==true)|.path]}',expected={release:ref.manifest.release,paths:ref.manifest.assets.filter(a=>a.publish).map(a=>a.path)},expectedFiles=Object.fromEntries(expected.paths.map(n=>[n,desc(Buffer.from(ref.assets[n]))]));
function tool(n){const r=cp.spawnSync(n,['--version'],{encoding:'utf8',timeout:5000});return {status:r.status,error:r.error?.code??null,version:r.stdout?.split('\n')[0]??null}}
function before(){const files=Object.fromEntries(Object.keys(ref.files).map(n=>[n,desc(fs.readFileSync(n))]));assert.deepEqual(files,ref.files);const input=fs.readFileSync('qa-assets-input.json');assert.deepEqual(JSON.parse(input),ref.manifest);for(const [n,s]of Object.entries(ref.assets))assert.deepEqual(fs.readFileSync(path.join('qa-assets-src',n)),Buffer.from(s));return {marker,node:process.version,files,program_sha256:programSHA,input:desc(input),tools:Object.fromEntries(['jq','rsync'].map(n=>[n,tool(n)]))}}
function copied(root){const out={};function walk(dir,rel=''){for(const e of fs.readdirSync(dir,{withFileTypes:true})){const n=rel?rel+'/'+e.name:e.name;if(e.isDirectory())walk(path.join(dir,e.name),n);else{assert(e.isFile());out[n]=desc(fs.readFileSync(path.join(dir,e.name)))}}}walk(root);assert.deepEqual(out,expectedFiles);return out}
if(phase==='before'){for(const[n,s]of Object.entries(ref.assets)){const f=path.join('qa-assets-src',n);fs.mkdirSync(path.dirname(f),{recursive:true});fs.writeFileSync(f,s)}fs.writeFileSync('qa-assets-input.json',JSON.stringify(ref.manifest));fs.mkdirSync('dist',{recursive:true});fs.writeFileSync('dist/old-owned.txt','own stale output');console.log('QA_ASSET_BEFORE '+JSON.stringify(before()))}
else if(phase==='after-jq'){const v=JSON.parse(fs.readFileSync('qa-assets-filtered.json','utf8'));assert.deepEqual(v,expected);console.log('QA_ASSET_JQ '+JSON.stringify({...before(),manifest:v}))}
else if(phase==='build'){const manifest=JSON.parse(fs.readFileSync('qa-assets-filtered.json','utf8'));assert.deepEqual(manifest,expected);const v={...before(),manifest,copied_files:copied('dist')};fs.writeFileSync('qa-assets-build.json',JSON.stringify(v));console.log('QA_ASSET_AFTER '+JSON.stringify(v))}
else{const build=JSON.parse(fs.readFileSync('qa-assets-build.json','utf8'));http.createServer((req,res)=>{let tmp;try{tmp=fs.mkdtempSync(path.join(os.tmpdir(),'qa-assets-'));fs.writeFileSync(path.join(tmp,'old-owned.txt'),'own stale runtime output');const j=cp.spawnSync('jq',['-e','-c',filter,'qa-assets-input.json'],{encoding:'utf8',timeout:5000});assert.equal(j.status,0,j.stderr);const manifest=JSON.parse(j.stdout);assert.deepEqual(manifest,expected);const r=cp.spawnSync('rsync',['-a','--delete','--exclude=*.tmp','qa-assets-src/',tmp+'/'],{encoding:'utf8',timeout:5000});assert.equal(r.status,0,r.stderr);const v={...before(),manifest,copied_files:copied(tmp),request_url:req.url};const raw=req.url.split('?')[0]==='/copied-index'?fs.readFileSync(path.join(tmp,'index.html')):Buffer.from(JSON.stringify({build,runtime:v}));res.writeHead(200,{'content-type':req.url.startsWith('/copied-index')?'text/html; charset=utf-8':'application/json','content-length':raw.length});res.end(raw);console.log('QA_ASSET_HTTP '+JSON.stringify({marker,path:req.url,copied_files:v.copied_files,manifest}))}catch(e){res.writeHead(500);res.end('own fixture failure');console.error(e.stack)}finally{if(tmp){assert(tmp.startsWith(path.join(os.tmpdir(),'qa-assets-')));fs.rmSync(tmp,{recursive:true,force:true})}}}).listen(Number(process.env.PORT||10000),'0.0.0.0')}
