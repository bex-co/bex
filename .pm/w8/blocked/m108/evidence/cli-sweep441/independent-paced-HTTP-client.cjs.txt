const fs=require('node:fs'),http=require('node:http'),https=require('node:https'),crypto=require('node:crypto');
const [url,file,stepText,receipt]=process.argv.slice(2),u=new URL(url),payload=fs.readFileSync(file),step=Number(stepText),start=process.hrtime.bigint();
const elapsed=()=>Number(process.hrtime.bigint()-start)/1e9;let sent=0,done=false,timer,deadline;
const out={client:process.version,url,bytes:payload.length,sha256:crypto.createHash('sha256').update(payload).digest('hex'),chunk_bytes:4096,interval_ms:step,writes:[],errors:[]};
const save=()=>{const t=receipt+'.tmp';fs.writeFileSync(t,JSON.stringify(out,null,2),{mode:0o600});fs.renameSync(t,receipt)};
function finish(){if(done)return;done=true;clearTimeout(timer);clearTimeout(deadline);out.finished_at=elapsed();out.bytes_submitted=sent;out.complete=true;save();if(!request.writableEnded)request.destroy();console.log(JSON.stringify({HTTP:out.response?.status,elapsed:out.finished_at,submitted:sent,response_complete:out.response?.complete,error:out.errors}));}
const request=(u.protocol==='https:'?https:http).request(u,{method:'PUT',headers:{'Content-Length':payload.length,'Content-Type':'application/octet-stream','User-Agent':'qa441-owned-file-transfer'}},res=>{out.response={status:res.statusCode,headers:res.headers,started_at:elapsed(),body:''};save();res.setEncoding('utf8');res.on('data',s=>{out.response.body+=s;save()});res.on('end',()=>{out.response.complete=true;try{out.response.decoded=JSON.parse(out.response.body)}catch{}finish()});res.on('error',e=>{out.errors.push({phase:'response',code:e.code,message:e.message});finish()})});request.on('error',e=>{out.errors.push({phase:'request',code:e.code,message:e.message});if(!out.response||out.response.complete)finish();else save()});request.flushHeaders();deadline=setTimeout(()=>{out.client_deadline_reached=true;request.destroy(new Error('Own120s client deadline'));finish()},120000);
function send(){if(done)return;if(sent===payload.length){out.upload_end_at=elapsed();save();request.end();return}const end=Math.min(sent+4096,payload.length),ok=request.write(payload.subarray(sent,end));sent=end;out.writes.push({at:elapsed(),bytes:sent,backpressure:!ok});save();if(ok)timer=setTimeout(send,step);else request.once('drain',()=>{timer=setTimeout(send,step)})}save();send();
