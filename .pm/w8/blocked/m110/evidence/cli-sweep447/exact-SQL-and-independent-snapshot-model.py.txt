def snapshots():
 rows={1:['north',1250],2:['north',725],3:['café:李',1999],4:['café:李',-499],5:['west',0]};out=[]
 for phase in ['initial','insert','update','delete']:
  if phase=='insert':rows[6]=['north',300]
  if phase=='update':rows[2]=['south',-125]
  if phase=='delete':del rows[4]
  groups={}
  for region,cents in rows.values():count,total=groups.get(region,(0,0));groups[region]=(count+1,total+cents)
  out.append({'phase':phase,'rows':sorted([[r,*v] for r,v in groups.items()])})
 return out

 view="SELECT region,count(*) AS cnt,sum(cents) AS total FROM "+schema+".orders GROUP BY region";out={'catalog':"SELECT jsonb_build_object('version',current_setting('server_version_num'),'encoding',current_setting('server_encoding'),'available',COALESCE((SELECT jsonb_agg(jsonb_build_object('name',name,'default',default_version,'installed',installed_version) ORDER BY name) FROM pg_available_extensions WHERE name IN ('pg_ivm','vector','pg_stat_statements','plpgsql','hstore')),'[]'::jsonb),'own_schema_exists',EXISTS(SELECT FROM pg_namespace WHERE nspname='"+schema+"'))::text;",'enable':'CREATE EXTENSION IF NOT EXISTS pg_ivm;'}

  sql=begin
  if case=='ivm':sql+="SELECT pgivm.create_immv('"+schema+".totals', '"+view+"'); "
  selected=schema+'.totals' if case=='ivm' else '('+view+') q'

 return out
