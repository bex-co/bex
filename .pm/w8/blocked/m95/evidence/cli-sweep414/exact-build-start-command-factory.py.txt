import json,hashlib
def commands(P,reference,marker,control=False):
 build='set -e\n';program=(P/'sweep414-app.py').read_text()
 for name,value in [('qa-vips-app.py',program),('qa-vips-reference.json',json.dumps(reference,separators=(',',':'),ensure_ascii=False)),('qa-vips-marker.json',json.dumps(marker))]:
  tag='QA_VIPS_'+name.replace('.','_').replace('-','_').upper();build+="cat > "+name+" <<'"+tag+"'\n"+value+('' if value.endswith('\n') else '\n')+tag+'\n'
 if control:build+='apt-get update -qq && apt-get install -y --no-install-recommends libvips42\n'
 build+='pip install --disable-pip-version-check pyvips==3.2.0\npython qa-vips-app.py build'
 return {'build':build,'start':'python -u qa-vips-app.py serve','program_sha256':hashlib.sha256(program.encode()).hexdigest()}
