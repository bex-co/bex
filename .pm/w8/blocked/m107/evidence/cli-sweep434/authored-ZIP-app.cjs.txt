const fs=require('node:fs'),os=require('node:os'),path=require('node:path'),cp=require('node:child_process'),http=require('node:http'),crypto=require('node:crypto');
const marker=JSON.parse(fs.readFileSync('qa434-marker.json')),reference=JSON.parse(fs.readFileSync('qa434-reference.json'));
const hash=b=>crypto.createHash('sha256').update(b).digest('hex');
function source(){const v=Object.fromEntries(Object.keys(reference.files).map(n=>{const b=fs.readFileSync(n);return[n,{bytes:b.length,sha256:hash(b)}]}));if(JSON.stringify(v)!==JSON.stringify(reference.files))throw Error('original source bytes changed');return v}
function archive(label){const dir=fs.mkdtempSync(path.join(os.tmpdir(),'qa434-'));try{const values={'LICENSE':fs.readFileSync('LICENSE'),'CHANGELOG':fs.readFileSync('CHANGELOG'),'greeting.txt':Buffer.from('Crème brûlée & 李\n'.repeat(7)),'two words.bin':Buffer.from(Array.from({length:8192},(_,i)=>(i*31+17)&255)),'empty.txt':Buffer.alloc(0),'request.txt':Buffer.from(label)};const names=Object.keys(values).sort(),manifest={};for(const n of names){fs.writeFileSync(path.join(dir,n),values[n]);fs.utimesSync(path.join(dir,n),946684800,946684800);manifest[n]={bytes:values[n].length,sha256:hash(values[n])}}
 const zip=path.join(dir,'out.zip');cp.execFileSync('zip',['-X','-9','-q',zip,...names],{cwd:dir,timeout:8000});const test=cp.execFileSync('unzip',['-t',zip],{encoding:'utf8',timeout:5000});if(!test.includes('No errors detected'))throw Error('actual unzip test failed');const bytes=fs.readFileSync(zip);return {manifest,archive:{bytes:bytes.length,sha256:hash(bytes),base64:bytes.toString('base64')},unzip_test:test.trim()}
 }finally{fs.rmSync(dir,{recursive:true,force:true})}}
function zipVersion(){return cp.execFileSync('zip',['-v'],{encoding:'utf8',timeout:5000}).split('\n').find(s=>s.startsWith('This is Zip '))}
function snapshot(phase,label){return {marker,phase,node:process.version,source:source(),...archive(label),zip:zipVersion()}}
if(process.argv[2]==='build'){console.log('QA434_BEFORE '+JSON.stringify({marker,node:process.version,source:source(),unzip:cp.execFileSync('unzip',['-v'],{encoding:'utf8'}).split('\n')[0]}));const v=snapshot('build',marker+'|build');fs.writeFileSync('qa434-built.json',JSON.stringify(v));console.log('QA434_BUILD '+JSON.stringify(v))}
else if(process.argv[2]==='serve'){const built=JSON.parse(fs.readFileSync('qa434-built.json')),boot=crypto.randomUUID();if(JSON.stringify(built.source)!==JSON.stringify(source()))throw Error('build/runtime source mismatch');console.log('QA434_RUNTIME '+JSON.stringify({marker,boot,node:process.version,zip:zipVersion()}));http.createServer((q,r)=>{try{const u=new URL(q.url,'http://local'),nonce=u.searchParams.get('qa')??'root';if(nonce.length>100)throw Error('nonce too long');const runtime=snapshot('runtime',marker+'|'+nonce);console.log('QA434_HTTP '+JSON.stringify({marker,boot,nonce,archive_sha256:runtime.archive.sha256}));if(u.pathname==='/download'){const b=Buffer.from(runtime.archive.base64,'base64');r.writeHead(200,{'Content-Type':'application/zip','Content-Disposition':'attachment; filename="qa434.zip"','Content-Length':b.length});r.end(b)}else{r.writeHead(200,{'Content-Type':'application/json'});r.end(JSON.stringify({marker,boot,nonce,built,runtime}))}}catch(e){r.writeHead(500,{'Content-Type':'text/plain'});r.end(String(e))}}).listen(Number(process.env.PORT||3000),'0.0.0.0')}
else throw Error('bad mode');
