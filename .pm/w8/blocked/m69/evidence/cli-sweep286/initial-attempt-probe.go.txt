func TestQA286NativeCronCLIUpdate(t *testing.T) {
 dir:=os.Getenv("QA286_PROBE_DIR")
 raw,err:=os.ReadFile(filepath.Join(dir,"sweep285-cron-update-wire.json"));if err!=nil {t.Fatal(err)}
 var capture struct{Cases []struct{Client string `json:"client"`;Authored string `json:"authored"`;Requests []struct{Method string `json:"method"`;Body json.RawMessage `json:"body"`} `json:"requests"`} `json:"cases"`}
 if err=json.Unmarshal(raw,&capture);err!=nil {t.Fatal(err)}
 for _,row:=range capture.Cases {t.Run(row.Client,func(t *testing.T){
  appRaw,err:=os.ReadFile(filepath.Join(dir,"sweep285-initial-proposal-produced","sweep285-app-"+row.Client+".json"));if err!=nil {t.Fatal(err)}
  var app appv1alpha1.App;if err=json.Unmarshal(appRaw,&app);err!=nil {t.Fatal(err)}
  app.Spec.StartCommand="printf '%s' QA286_OLD";app.Spec.Command=""
  svc,cl:=newService(nil,&app);mux:=http.NewServeMux();svc.RegisterREST(mux)
  var body json.RawMessage;for _,r:=range row.Requests {if r.Method=="PATCH" {body=r.Body}}
  if len(body)==0 {t.Fatal("actual native patch missing")}
  rec:=httptest.NewRecorder();mux.ServeHTTP(rec,httptest.NewRequest("PATCH","/v1/services/qa285",strings.NewReader(string(body))))
  if rec.Code!=http.StatusOK {t.Fatalf("PATCH %d: %s",rec.Code,rec.Body)}
  after:=getApp(t,cl,"qa285");record,_:=json.Marshal(map[string]any{"kind":"native CLI update","subject":row.Client,"request":json.RawMessage(body),"expected_start":row.Authored,"actual_start":after.Spec.StartCommand,"actual_override":after.Spec.Command,"response_status":rec.Code,"response_body":rec.Body.String(),"native_role_preserved":after.Spec.Command==""&&after.Spec.StartCommand==row.Authored});t.Log("QA286_TRANSITION "+string(record))
  if after.Spec.Command!=""||after.Spec.StartCommand!=row.Authored {t.Error("native start update became an image override instead of a native start")}
 })}
}

func TestQA286NativeCronBlueprintEdit(t *testing.T) {
 oldCommand:="printf '%s' QA286_OLD";wanted:="[[ 1 = 1 ]] && printf '%s' QA286_NEW"
 manifest:="services:\n  - name: qa286-cron\n    type: cron\n    runtime: node\n    repo: https://github.com/bex-co/bex\n    schedule: '0 0 1 1 *'\n    buildCommand: /bin/true\n    startCommand: "+oldCommand+"\n"
 rec:=&recordingStore{};svc,cl:=newService(rec)
 if _,err:=svc.DeployStack(t.Context(),DeployRequest{Manifest:manifest});err!=nil {t.Fatal(err)}
 baseline:=serializedBlueprintApp(t,manage(getApp(t,cl,"qa286-cron"),id.New(id.Service)))
 if baseline.Spec.Command!=""||baseline.Spec.StartCommand!=oldCommand {t.Fatal("initial native role is wrong")}
 svc.Clock=func() time.Time{return time.Date(2026,10,15,0,0,0,0,time.UTC)}
 svc.Client=fakeClient(serializedBlueprintApp(t,baseline))
 if _,err:=svc.DeployStack(t.Context(),DeployRequest{Manifest:manifest});err!=nil {t.Fatal(err)}
 unchanged:=getApp(t,svc.Client,"qa286-cron");if unchanged.Spec.Command!=""||unchanged.Spec.StartCommand!=oldCommand {t.Fatal("unchanged native control changed role")}
 if _,err:=svc.DeployStack(t.Context(),DeployRequest{Manifest:strings.Replace(manifest,oldCommand,wanted,1)});err!=nil {t.Fatal(err)}
 after:=getApp(t,svc.Client,"qa286-cron");record,_:=json.Marshal(map[string]any{"kind":"actual Blueprint genuine command edit","initial_control":true,"unchanged_control":true,"expected_start":wanted,"actual_start":after.Spec.StartCommand,"actual_override":after.Spec.Command,"native_role_preserved":after.Spec.Command==""&&after.Spec.StartCommand==wanted,"deploy_rows":len(rec.deployCalls)});t.Log("QA286_TRANSITION "+string(record))
 if after.Spec.Command!=""||after.Spec.StartCommand!=wanted {t.Error("genuine native command edit became an image override")}
}
