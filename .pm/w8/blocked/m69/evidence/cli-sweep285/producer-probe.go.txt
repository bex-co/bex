package apps

import (
 "context"
 "encoding/json"
 "os"
 "path/filepath"
 "testing"
 appv1alpha1 "github.com/bex-co/bex/lego/types/v1alpha1"
)

func TestQA285NativeCronProducer(t *testing.T) {
 dir:=os.Getenv("QA285_PROBE_DIR");if dir=="" {t.Fatal("private probe directory required")}
 raw,err:=os.ReadFile(filepath.Join(dir,"sweep285-cron-command-wire.json"));if err!=nil {t.Fatal(err)}
 var all struct{Cases []struct{Client string `json:"client"`;Requests []struct{Method string `json:"method"`;Body json.RawMessage `json:"body"`} `json:"requests"`} `json:"cases"`}
 if err=json.Unmarshal(raw,&all);err!=nil {t.Fatal(err)}
 if len(all.Cases)!=2 {t.Fatal("two actual native captures required")}
 for _,row:=range all.Cases {t.Run(row.Client,func(t *testing.T){
  var body json.RawMessage
  for _,request:=range row.Requests {if request.Method=="POST" {body=request.Body}}
  if len(body)==0 {t.Fatal("native POST body missing")}
  var transport createServiceRequest;if err:=json.Unmarshal(body,&transport);err!=nil {t.Fatal(err)}
  req,err:=transport.toCreateRequest(context.Background(),"");if err!=nil {t.Fatal(err)}
  spec,err:=specFromCreate(req);if err!=nil {t.Fatal(err)}
  app:=&appv1alpha1.App{};app.Name="qa285";app.Namespace="default";app.Spec=spec
  encoded,_:=json.MarshalIndent(app,"","  ")
  if err=os.WriteFile(filepath.Join(dir,"sweep285-app-"+row.Client+".json"),encoded,0600);err!=nil {t.Fatal(err)}
  t.Logf("runtime=%s builder=%s start=%q command=%q",spec.Runtime,spec.Builder,spec.StartCommand,spec.Command)
  if spec.Runtime!="node"||spec.Builder!="native"||spec.StartCommand=="" {t.Fatal("native cron producer fixture is invalid")}
 })}
}
