const http=require('node:http'),https=require('node:https'),zlib=require('node:zlib');
const [url,marker,offer]=process.argv.slice(2),start=performance.now();let done=false,text='',rows=[],chunks=[],record={url,offer};
const finish=(error)=>{if(done)return;done=true;clearTimeout(timer);record={...record,error:error||null,elapsed_ms:Math.round(performance.now()-start),text,events:rows,chunks};record.pass=!error&&record.status===200&&rows.length===3&&rows.every((r,i)=>r.value.marker===marker&&r.value.seq===i&&r.value.text==='café 日本語')&&rows[2].at_ms-rows[0].at_ms>=1500;console.log(JSON.stringify(record));process.exitCode=error?1:0;};
const req=(url.startsWith('https:')?https:http).get(url,{headers:{'Accept':'text/event-stream','Accept-Encoding':offer,'User-Agent':'bex-functional-qa/352'}},res=>{
 record.status=res.statusCode;record.headers=res.headers;record.headers_ms=Math.round(performance.now()-start);
 let stream=res;const encoding=res.headers['content-encoding'];if(encoding==='gzip')stream=res.pipe(zlib.createGunzip());else if(encoding==='br')stream=res.pipe(zlib.createBrotliDecompress());else if(encoding==='deflate')stream=res.pipe(zlib.createInflate());else if(encoding&&encoding!=='identity'){res.resume();finish('unsupported response encoding '+encoding);return;}
 stream.setEncoding('utf8');let pending='';stream.on('data',chunk=>{const at=Math.round(performance.now()-start);chunks.push({at_ms:at,chars:chunk.length});text+=chunk;pending+=chunk;if(text.length>8192){req.destroy();finish('fixture response exceeds expected bound');return;}let end;while((end=pending.indexOf('\n\n'))>=0){const block=pending.slice(0,end);pending=pending.slice(end+2);const data=block.split('\n').find(s=>s.startsWith('data: '));if(data){try{rows.push({at_ms:at,value:JSON.parse(data.slice(6))});}catch(e){finish(String(e));}}}});stream.on('error',e=>finish(String(e)));stream.on('end',()=>finish());
});req.on('error',e=>finish(String(e)));const timer=setTimeout(()=>{req.destroy();finish('bounded 12s observation timed out');},12000);
