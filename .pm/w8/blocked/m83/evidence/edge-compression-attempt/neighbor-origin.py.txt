import gzip, http.server, socketserver, time, hashlib
TEXT = ("bex compression probe line " * 40 + "\n").encode() * 300  # ~300KB
GZ = gzip.compress(TEXT, mtime=0)
ETAG = '"' + hashlib.sha256(TEXT).hexdigest()[:16] + '"'
class H(http.server.BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"
    def log_message(self, *a): pass
    def send(self, code, body, headers):
        self.send_response(code)
        for k, v in headers.items(): self.send_header(k, v)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        if self.command != "HEAD": self.wfile.write(body)
    def do_HEAD(self): self.do_GET()
    def do_GET(self):
        p = self.path
        if p == "/ws":
            import base64
            key = self.headers.get("Sec-WebSocket-Key", "")
            acc = base64.b64encode(hashlib.sha1((key + "258EAFA5-E914-47DA-95CA-C5AB0DC85B11").encode()).digest()).decode()
            self.send_response(101); self.send_header("Upgrade", "websocket"); self.send_header("Connection", "Upgrade"); self.send_header("Sec-WebSocket-Accept", acc); self.end_headers(); self.wfile.flush()
            data = self.rfile.read1(4096) if hasattr(self.rfile, "read1") else self.connection.recv(4096)
            self.wfile.write(b"\x81\x0bws-echo-ok!"); self.wfile.flush()
            return
        if p == "/text": self.send(200, TEXT, {"Content-Type": "text/plain; charset=utf-8"})
        elif p == "/json": self.send(200, b'{"a":"' + b"x" * 5000 + b'"}', {"Content-Type": "application/json"})
        elif p == "/small": self.send(200, b"tiny body", {"Content-Type": "text/plain"})
        elif p == "/gzip": self.send(200, GZ, {"Content-Type": "text/plain", "Content-Encoding": "gzip"})
        elif p == "/notransform": self.send(200, TEXT, {"Content-Type": "text/plain", "Cache-Control": "no-transform"})
        elif p == "/png": self.send(200, b"\x89PNG" + bytes(range(256)) * 40, {"Content-Type": "image/png"})
        elif p == "/notransform-etag": self.send(200, TEXT, {"Content-Type": "text/plain", "Cache-Control": "public, No-Transform", "ETag": ETAG})
        elif p == "/forged": self.send(200, TEXT, {"Content-Type": "text/plain", "X-Bex-Compress-Held": "1", "X-Bex-Compress-Plain": "1"})
        elif p == "/weak-etag": self.send(200, TEXT, {"Content-Type": "text/plain", "ETag": "W/" + ETAG})
        elif p == "/svg": self.send(200, b"<svg xmlns='http://www.w3.org/2000/svg'>" + b"<g/>" * 2000 + b"</svg>", {"Content-Type": "image/svg+xml"})
        elif p == "/etag":
            if self.headers.get("If-None-Match") in (ETAG, "W/" + ETAG): self.send(304, b"", {"ETag": ETAG})
            else: self.send(200, TEXT, {"Content-Type": "text/plain", "ETag": ETAG})
        elif p == "/range":
            r = self.headers.get("Range")
            if r and r.startswith("bytes=0-99"): self.send(206, TEXT[:100], {"Content-Type": "text/plain", "Content-Range": "bytes 0-99/%d" % len(TEXT), "Accept-Ranges": "bytes"})
            else: self.send(200, TEXT, {"Content-Type": "text/plain", "Accept-Ranges": "bytes"})
        elif p in ("/stream-text", "/stream-ndjson", "/stream-small"):
            ct = {"/stream-text": "text/plain", "/stream-ndjson": "application/x-ndjson", "/stream-small": "text/plain"}[p]
            self.send_response(200); self.send_header("Content-Type", ct); self.send_header("Transfer-Encoding", "chunked"); self.end_headers()
            for i in range(3):
                pad = "z" * (2000 if p != "/stream-small" else 10)
                data = ('{"token": %d, "pad": "%s"}\n' % (i, pad)).encode()
                self.wfile.write(b"%x\r\n%s\r\n" % (len(data), data)); self.wfile.flush(); time.sleep(1)
            self.wfile.write(b"0\r\n\r\n")
        elif p == "/sse":
            self.send_response(200); self.send_header("Content-Type", "text/event-stream"); self.send_header("Cache-Control", "no-cache"); self.send_header("Transfer-Encoding", "chunked"); self.end_headers()
            for i in range(3):
                data = ("data: event-%d %s\n\n" % (i, "y" * 2000)).encode()
                self.wfile.write(b"%x\r\n%s\r\n" % (len(data), data)); self.wfile.flush(); time.sleep(1)
            self.wfile.write(b"0\r\n\r\n")
        else: self.send(404, b"nf", {"Content-Type": "text/plain"})
class T(socketserver.ThreadingMixIn, http.server.HTTPServer): daemon_threads = True
T(("0.0.0.0", 8000), H).serve_forever()
