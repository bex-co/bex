const fs=require('node:fs'),cp=require('node:child_process'),crypto=require('node:crypto'),http=require('node:http'),ref=require('./qa402-reference.json'),marker=require('./qa402-marker.json'),phase=process.argv[2],sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const files=Object.fromEntries(Object.keys(ref.files).map(p=>{const b=fs.readFileSync(p);return[p,{bytes:b.length,sha256:sha(b)}]}));if(JSON.stringify(files)!==JSON.stringify(ref.files))throw Error('source differs');const b=fs.readFileSync('qa402.dump');if(b.length!==ref.backup.bytes||sha(b)!==ref.backup.sha256)throw Error('backup differs');
const tools=Object.fromEntries(['psql','pg_dump','pg_restore'].map(p=>{const r=cp.spawnSync(p,['--version'],{encoding:'utf8'});return[p,{status:r.status,error:r.error?.code??null,stdout:r.stdout??'',stderr:r.stderr??''}]}));const before={marker,node:process.version,files,backup:ref.backup,tools};
if(phase==='before'){fs.writeFileSync('qa402-before.json',JSON.stringify(before));console.log('QA402_BEFORE '+JSON.stringify(before));process.exit(0)}
function restore(){const r=cp.spawnSync('pg_restore',['--data-only','--no-owner','--no-privileges','--file=-','qa402.dump'],{encoding:'utf8'});if(r.error||r.status!==0)throw Error('pg_restore failed: '+(r.error?.code??r.stderr));const rows=r.stdout.split('\n').filter(s=>s.startsWith('INSERT INTO public.qa402_backup VALUES'));if(JSON.stringify(rows)!==JSON.stringify(ref.rows)||sha(rows.join('\n')+'\n')!==ref.rows_sha256)throw Error('restored rows differ');return{marker,node:process.version,files,backup:ref.backup,tools,rows,rows_sha256:ref.rows_sha256,restored_from_archive:true,before:JSON.parse(fs.readFileSync('qa402-before.json'))}}
if(phase==='build'){const v=restore();fs.writeFileSync('qa402-build.json',JSON.stringify(v));console.log('QA402_AFTER '+JSON.stringify(v))}else{http.createServer((req,res)=>{if(req.url.split('?')[0]!=='/'&&req.url.split('?')[0]!=='/restore'){res.writeHead(404);return res.end()}try{const runtime={...restore(),request_url:req.url};res.setHeader('Content-Type','application/json');res.end(JSON.stringify({build:require('./qa402-build.json'),runtime}));console.log('QA402_RUNTIME '+JSON.stringify(runtime))}catch(e){res.writeHead(500);res.end(String(e))}}).listen(Number(process.env.PORT),'0.0.0.0')}
