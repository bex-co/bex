def cli(args):
    if client=='bex':
        return json.loads(subprocess.check_output(['python3',str(p/'run.py'),'--release',*args],text=True))
    cfg=p/'render-config/cli.yaml'
    assert not cfg.exists()
    cfg.parent.mkdir(mode=0o700,exist_ok=True)
    shutil.copyfile(p/'config/cli.yaml',cfg)
    cfg.chmod(0o600)
    try: return json.loads(subprocess.check_output(['python3',str(p/'run-render.py'),*args],text=True))
    finally: cfg.unlink(missing_ok=True)
