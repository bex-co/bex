import json,gzip,base64,hashlib
def commands(P,reference,marker,control=False):
 files={dst:(P/src).read_text() for dst,src in [('qa-webpack-entry.mjs','candidate-webpack-entry.mjs'),('qa-webpack-domain.mjs','candidate-webpack-domain.mjs'),('qa-webpack-before.cjs','candidate-webpack-before.cjs')]};assert all(hashlib.sha256(files[n].encode()).hexdigest()==v for n,v in reference['sources'].items());files.update({'qa-webpack-reference.json':json.dumps(reference,separators=(',',':'),ensure_ascii=False),'qa-webpack-marker.json':json.dumps(marker)});blob=base64.b64encode(gzip.compress(json.dumps(files,ensure_ascii=False,separators=(',',':')).encode(),mtime=0)).decode();assert json.loads(gzip.decompress(base64.b64decode(blob)))==files
