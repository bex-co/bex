const fs=require('node:fs'),crypto=require('node:crypto'),child=require('node:child_process');const ref=JSON.parse(fs.readFileSync('qa-webpack-reference.json','utf8')),marker=JSON.parse(fs.readFileSync('qa-webpack-marker.json','utf8'));const sha=b=>crypto.createHash('sha256').update(b).digest('hex');const files=Object.fromEntries(Object.keys(ref.files).map(name=>{const b=fs.readFileSync(name);return [name,{bytes:b.length,sha256:sha(b)}]}));if(JSON.stringify(files)!==JSON.stringify(ref.files))throw Error('public source changed');const sources=Object.fromEntries(Object.keys(ref.sources).map(name=>[name,sha(fs.readFileSync(name))]));if(JSON.stringify(sources)!==JSON.stringify(ref.sources))throw Error('own source changed');const t=child.spawnSync('webpack',['--version'],{encoding:'utf8'});console.log('QA_WEBPACK_BEFORE',JSON.stringify({marker,node:process.version,files,sources,webpack_probe:{status:t.status,error:t.error?.code||null,stdout:t.stdout||null}}));
