type qa77ObservedWrite struct {
	Health            string `json:"health"`
	PreDeploy         string `json:"predeploy"`
	Start             string `json:"start"`
	ReleaseStamp      string `json:"releaseStamp"`
	DeployRowsAtWrite int    `json:"deployRowsAtWrite"`
}
type qa77ObservingClient struct {
	client.Client
	store  *recordingStore
	writes []qa77ObservedWrite
}

func (o *qa77ObservingClient) Patch(ctx context.Context, obj client.Object, patch client.Patch, opts ...client.PatchOption) error {
	po := &client.PatchOptions{}
	for _, opt := range opts {
		opt.ApplyToPatch(po)
	}
	if err := o.Client.Patch(ctx, obj, patch, opts...); err != nil {
		return err
	}
	if len(po.DryRun) > 0 {
		return nil
	}
	if a, ok := obj.(*appv1alpha1.App); ok {
		fresh := &appv1alpha1.App{}
		if err := o.Client.Get(ctx, client.ObjectKeyFromObject(a), fresh); err != nil {
			return err
		}
		o.writes = append(o.writes, qa77ObservedWrite{Health: fresh.Spec.HealthCheckPath, PreDeploy: fresh.Spec.PreDeployCommand, Start: fresh.Spec.StartCommand, ReleaseStamp: fresh.Annotations[appv1alpha1.AnnotationReleaseGeneration], DeployRowsAtWrite: len(o.store.deployCalls)})
	}
	return nil
}
func TestQA77PersistedCompoundConfigurations(t *testing.T) {
	empty := ""
	start := "./new-app"
	cases := []struct {
		name  string
		patch ServicePatch
	}{
		{"health-and-predeploy-clear", ServicePatch{HealthCheckPath: &empty, PreDeployCommand: &empty}},
		{"single-predeploy-clear", ServicePatch{PreDeployCommand: &empty}},
		{"predeploy-clear-and-start-change", ServicePatch{PreDeployCommand: &empty, StartCommand: &start}},
	}
	var results []map[string]any
	for _, tc := range cases {
		st := &recordingStore{}
		a := managedRepoApp("qa77-web")
		a.Spec.HealthCheckPath = "/healthz"
		a.Spec.PreDeployCommand = "printf qa77-prior"
		svc, cl := newService(st, a)
		observer := &qa77ObservingClient{Client: cl, store: st}
		svc.Client = observer
		_, err := svc.ApplyServicePatch(context.Background(), a.Name, tc.patch)
		if err != nil {
			t.Fatalf("%s: %v", tc.name, err)
		}
		final := getApp(t, cl, a.Name)
		result := map[string]any{"case": tc.name, "writes": observer.writes, "deployRowsAfterReturn": len(st.deployCalls), "finalHealth": final.Spec.HealthCheckPath, "finalPreDeploy": final.Spec.PreDeployCommand, "fakeClientGenerationDoesNotAdvance": true, "operatorOrKubernetesExecution": false}
		results = append(results, result)
		raw, _ := json.Marshal(result)
		t.Log(string(raw))
	}
	raw, err := json.MarshalIndent(results, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(filepath.Join(os.Getenv("BEX_QA77_DIR"), "sweep77-persisted-observations.json"), append(raw, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
}
