const fs=require('fs'),crypto=require('crypto'),cp=require('child_process'),path=require('path');
const reference=JSON.parse(fs.readFileSync('qa431-reference.json')),marker=JSON.parse(fs.readFileSync('qa431-marker.json'));
const hash=b=>({bytes:b.length,sha256:crypto.createHash('sha256').update(b).digest('hex')});
const files=Object.fromEntries(Object.keys(reference.files).map(n=>[n,hash(fs.readFileSync(n))]));if(JSON.stringify(files)!==JSON.stringify(reference.files))throw Error('original parent/theme bytes changed');
const index=fs.readFileSync('public/index.html');if(!index.includes(Buffer.from('Deploy Hugo on Render'))||!index.includes(Buffer.from('Hugo')))throw Error('actual Hugo index absent');
function walk(d){return fs.readdirSync(d,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(d,e.name)):[path.join(d,e.name)])}
const css=walk('public').filter(n=>n.endsWith('.css')).sort().map(n=>({path:'/'+path.relative('public',n),...hash(fs.readFileSync(n))}));if(!css.length)throw Error('actual theme stylesheet absent');
const v={marker,files,node:process.version,hugo:cp.execFileSync('hugo',['version'],{encoding:'utf8'}).trim(),index:hash(index),css};fs.writeFileSync('public/qa431-evidence.json',JSON.stringify(v));console.log('QA431_AFTER '+JSON.stringify(v));
