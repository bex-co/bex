const fs=require('node:fs'),crypto=require('node:crypto');
const WebSocket=require(process.argv[2]);
const [url,id,journal,durationArg,intervalArg]=process.argv.slice(3);
const duration=Number(durationArg),interval=Number(intervalArg);
if(!Number.isFinite(duration)||duration<1||duration>185||!Number.isFinite(interval)||interval<0.1)throw new Error('finite bounds required');
const started=Date.now();let opened=null,sent=0,received=0,pings=0,pongs=0,closing=false,done=false;const timers=[];
function log(e){fs.appendFileSync(journal,JSON.stringify({at:new Date().toISOString(),elapsed_ms:Date.now()-started,...e})+'\n',{mode:0o600});}
function finish(code,reason){if(done)return;done=true;for(const t of timers)clearTimeout(t);log({event:'FINAL',code,reason,opened:opened!==null,sent,received,pings,pongs,active_ms:opened===null?0:Date.now()-opened});process.exitCode=code;}
log({event:'INTENT',url,id,duration_seconds:duration,interval_seconds:interval,PID:process.pid});
const ws=new WebSocket(url+'?id='+encodeURIComponent(id),{handshakeTimeout:12000,perMessageDeflate:false});
function send(){if(ws.readyState!==WebSocket.OPEN||closing)return;const seq=sent++;const body=JSON.stringify({id,seq,at:new Date().toISOString(),payload:'incoming telemetry ✓ '+'.'.repeat(128)});const bytes=Buffer.from(body);ws.send(bytes,{binary:false},err=>{log({event:'SEND',seq,bytes:bytes.length,sha256:crypto.createHash('sha256').update(bytes).digest('hex'),error:err?err.message:null});});}
ws.on('open',()=>{opened=Date.now();log({event:'OPEN'});send();timers.push(setInterval(send,interval*1000));timers.push(setTimeout(()=>{closing=true;log({event:'NORMAL_CLOSE_INTENT',sent});ws.close(1000,'qa372 complete');},duration*1000));});
ws.on('message',data=>{received++;log({event:'UNEXPECTED_DATA',bytes:data.length});});
ws.on('ping',data=>{pings++;log({event:'PING',bytes:data.length});});
ws.on('pong',data=>{pongs++;log({event:'PONG',bytes:data.length});});
ws.on('error',err=>log({event:'ERROR',message:err.message}));
ws.on('close',(code,reason)=>{log({event:'CLOSE',code,reason:reason.toString(),normal_close_intent:closing});finish(closing&&code===1000&&received===0&&pings===0&&pongs===0?0:1,'socket closed');});
timers.push(setTimeout(()=>{log({event:'BOUND_EXHAUSTED'});ws.terminate();finish(2,'finite total deadline');},(duration+25)*1000));
process.on('SIGTERM',()=>{log({event:'HARNESS_ABORT'});ws.terminate();finish(3,'harness abort');});
