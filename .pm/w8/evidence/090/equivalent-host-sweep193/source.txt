11: // EnsureDashboardURL gives a config with no dashboard origin the Bex one
12: // before a command that opens dashboard pages runs (w8/090): upstream saves
13: // the origin only at device login, so a config that never logged in (a missing
14: // or version-only file, an environment credential) had none.
15: //
16: // It acts only for the production default API, and never replaces an origin
17: // already saved, so custom and local control planes and explicit overrides
18: // keep theirs. Only the non-secret origin is written; an environment bearer is
19: // never persisted. It is best-effort: an unreadable or unwritable config is
20: // left for upstream to report as it always has.
21: func EnsureDashboardURL(use bool) {
22: 	if !use || cfg.GetHost() != DefaultHost {
23: 		return
24: 	}
25: 	current, err := config.Load()
26: 	if err != nil || current.DashboardURL != "" || (current.Host != "" && current.Host != DefaultHost) {
27: 		return
28: 	}
29: 	current.DashboardURL = DefaultDashboardURL
30: 	_ = current.Persist()
31: }
