def cli(args, bound=45):
    cfg = P / 'render-config/cli.yaml'
    try:
        if client == 'render':
            assert not cfg.exists()
            shutil.copyfile(P / 'config/cli.yaml', cfg)
            cfg.chmod(0o600)
        runner = P / ('run-render.py' if client == 'render' else 'run.py')
        r = json.loads(subprocess.check_output(['python3', str(runner), *([] if client=='render' else ['--release']), '--bound=' + str(bound), *args], text=True, timeout=bound + 15))
        r['native_exit'] = None if r['exit'] == 124 else r['exit']
