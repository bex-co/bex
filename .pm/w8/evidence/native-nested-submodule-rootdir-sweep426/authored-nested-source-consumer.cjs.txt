const fs=require('node:fs'),crypto=require('node:crypto'),assert=require('node:assert/strict'),http=require('node:http');const ref=JSON.parse(fs.readFileSync('qa426-reference.json','utf8')),marker=JSON.parse(fs.readFileSync('qa426-marker.json','utf8'));const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
function observed(){const files=Object.fromEntries(Object.keys(ref.files).map(n=>{if(!fs.existsSync(n))return[n,null];const b=fs.readFileSync(n);return[n,{bytes:b.length,sha256:sha(b)}]}));const absent=Object.fromEntries(ref.absent.map(n=>[n,!fs.existsSync(n)]));return {marker,node:process.version,files,absent,program_sha256:sha(fs.readFileSync(__filename))};}
function verify(v){assert.deepEqual(v.files,ref.files);assert(Object.values(v.absent).every(Boolean));assert.equal(v.program_sha256,ref.program_sha256);return v;}
if(process.argv[2]==='build'){const v=observed();console.log('QA_SUBROOT_BEFORE',JSON.stringify(v));verify(v);fs.writeFileSync('qa426-build.json',JSON.stringify(v));console.log('QA_SUBROOT_AFTER',JSON.stringify(v));}
else if(process.argv[2]==='serve'){const build=JSON.parse(fs.readFileSync('qa426-build.json','utf8')),boot=crypto.randomUUID();http.createServer((req,res)=>{const runtime=verify(observed());runtime.path=req.url;res.writeHead(200,{'content-type':'application/json'});res.end(JSON.stringify({build,runtime,boot}));console.log('QA_SUBROOT_HTTP',JSON.stringify({marker,path:req.url,files:runtime.files,absent:runtime.absent,boot}));}).listen(Number(process.env.PORT||3000),'0.0.0.0');}
else throw Error('unknown mode');
