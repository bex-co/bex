use std::{env,fs,io::{Read,Write},net::TcpListener};
fn quote(s:&str)->String { let mut o=String::from("\""); for c in s.chars(){match c{'"'=>o.push_str("\\\""),'\\'=>o.push_str("\\\\"),'\n'=>o.push_str("\\n"),'\r'=>o.push_str("\\r"),'\t'=>o.push_str("\\t"),c if c<' '=>o.push_str(&format!("\\u{:04x}",c as u32)),c=>o.push(c)}} o.push('"');o }
fn snap()->String {let b=fs::read("/etc/secrets/qa-note.txt").unwrap();let h:String=b.iter().map(|v|format!("{:02x}",v)).collect();format!("{{\"marker\":{},\"path\":{},\"fileHex\":{}}}",quote(&env::var("QA_MARKER").unwrap()),quote(&env::var("PATH").unwrap()),quote(&h))}
fn main(){if env::args().nth(1).as_deref()==Some("--snapshot"){println!("{}",snap());return}let built=fs::read_to_string("qa-built.json").unwrap();let port=env::var("PORT").unwrap_or("3000".into());let listener=TcpListener::bind(format!("0.0.0.0:{}",port)).unwrap();for incoming in listener.incoming(){let mut client=incoming.unwrap();let mut req=[0u8;4096];if client.read(&mut req).is_err(){continue}let payload=format!("{{\"built\":{},\"runtime\":{}}}",built.trim(),snap());let response=format!("HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{}",payload.len(),payload);let _=client.write_all(response.as_bytes());}}
