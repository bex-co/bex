import("context";"encoding/json";"os";"testing";"github.com/bex-co/bex/lego/backend/internal/core")
func TestQAStaticLocation202(t *testing.T){raw,err:=os.ReadFile(os.Getenv("BEX_QA_LOCATION202_INPUT"));if err!=nil{t.Fatal(err)};var input struct{Cases []struct{Label string;Manifest string;ExpectedPointer string;ActualPath string}};if err=json.Unmarshal(raw,&input);err!=nil{t.Fatal(err)};for _,tc:=range input.Cases{t.Run(tc.Label,func(t *testing.T){source,problems:=CompileBlueprintSource(tc.Manifest);if len(problems)>0{t.Fatalf("compile:%+v",problems)};svc:=&Service{Base:&core.Base{Client:fakeClient(),Namespace:"default"}};got,err:=svc.ValidateBlueprint(context.Background(),"",tc.Manifest,"");if err!=nil||got.Valid||len(got.Errors)!=1{t.Fatalf("validate:%+v %v",got,err)};e:=got.Errors[0];wanted,ok:=source.Locations[tc.ExpectedPointer];if !ok||wanted.Line==0||e.Path==nil||*e.Path!=tc.ActualPath||e.Line==nil||*e.Line==wanted.Line{t.Fatalf("current mislocation not reproduced:%+v expected%+v",e,wanted)};b,_:=json.Marshal(map[string]any{"label":tc.Label,"actual":e,"expected_pointer":tc.ExpectedPointer,"expected_location":wanted,"expected_authored_pointer_exists":ok});t.Log("QA202_LOCATION "+string(b))})}}
