package controller

import (
	"context"
	"encoding/json"
	appv1alpha1 "github.com/bex-co/bex/lego/types/v1alpha1"
	corev1 "k8s.io/api/core/v1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"os"
	"path/filepath"
	"sigs.k8s.io/controller-runtime/pkg/client"
	"testing"
	"time"
)

func TestQA266RestartWakeProbe(t *testing.T) {
	for _, kind := range []string{"new_image", "plain_restart", "snapshot_restart", "snapshot_restart_saved"} {
		t.Run(kind, func(t *testing.T) {
			app := activeApp("qa266")
			app.Annotations[appv1alpha1.AnnotationReleaseGeneration] = "1"
			app.Spec.EnvFromSecret = app.Name + "-env"
			source := &corev1.Secret{ObjectMeta: metav1.ObjectMeta{Name: app.Spec.EnvFromSecret, Namespace: app.Namespace}, Data: map[string][]byte{"MESSAGE": []byte("V1")}}
			r, cl, nn := lifecycleFixture(t, app, withObjects(source))
			_, image := serveReleaseOne(t, r, cl, nn)
			updateApp(t, cl, nn, func(a *appv1alpha1.App) { a.Generation = 2; a.Spec.IdleTTLSeconds = 30 })
			if kind == "snapshot_restart_saved" {
				s := &corev1.Secret{}
				if err := cl.Get(context.Background(), client.ObjectKeyFromObject(source), s); err != nil {
					t.Fatal(err)
				}
				s.Data["MESSAGE"] = []byte("V2")
				if err := cl.Update(context.Background(), s); err != nil {
					t.Fatal(err)
				}
			}
			parkIdle(t, r, cl, nn)
			updateApp(t, cl, nn, func(a *appv1alpha1.App) {
				a.Generation = 3
				a.Annotations[appv1alpha1.AnnotationReleaseGeneration] = "3"
				a.Spec.RestartedAt = time.Now().UTC().Format(time.RFC3339Nano)
				if kind == "new_image" {
					a.Spec.Image = "nginx:2"
				}
				if kind == "snapshot_restart" || kind == "snapshot_restart_saved" {
					a.Spec.ReleaseConfig = &appv1alpha1.ReleaseConfigReference{Generation: 3, SourceGeneration: 1, Image: image, PreserveGroupValues: true}
				}
			})
			reconcileTwice(t, r, nn)
			live := liveApp(t, cl, nn)
			replicas := deploymentReplicas(t, cl, nn)
			t.Logf("kind=%s phase=%s replicas=%d release=%d served=%d wake=%q active=%s newer=%v", kind, live.Status.Phase, replicas, releaseGeneration(&live), successfulReleaseGeneration(&live), live.Annotations[annotReleaseWakeGeneration], live.Status.ActiveRevision, newerReleaseUnserved(&live))
			if replicas != 1 {
				t.Errorf("new restart must wake the sleeping service; got replicas=%d", replicas)
			}
			setDeploymentStatus(t, cl, nn, statusServedPodReady)
			reconcileTwice(t, r, nn)
			setDeploymentStatus(t, cl, nn, statusRolledOut)
			reconcileTwice(t, r, nn)
			served := liveApp(t, cl, nn)
			if served.Status.ActiveRevision != "rev-3" || served.Status.Phase != appv1alpha1.PhaseRunning {
				t.Fatalf("release did not serve: phase=%s active=%s", served.Status.Phase, served.Status.ActiveRevision)
			}
			parkIdle(t, r, cl, nn)
			parked := liveApp(t, cl, nn)
			if parked.Status.ActiveRevision != "rev-3" || parked.Status.ReleaseGeneration != 3 {
				t.Fatalf("park lost success: release=%d active=%s", parked.Status.ReleaseGeneration, parked.Status.ActiveRevision)
			}
			raw, err := json.MarshalIndent(parked, "", "  ")
			if err != nil {
				t.Fatal(err)
			}
			dir := os.Getenv("QA266_PROBE_DIR")
			if dir == "" {
				t.Fatal("QA266_PROBE_DIR required")
			}
			if err := os.WriteFile(filepath.Join(dir, "sweep266-parked-"+kind+".json"), raw, 0600); err != nil {
				t.Fatal(err)
			}
			t.Logf("parked producer state: kind=%s phase=%s release=%d active=%s", kind, parked.Status.Phase, parked.Status.ReleaseGeneration, parked.Status.ActiveRevision)

		})
	}
}
