package store

import (
	"encoding/json"
	appv1alpha1 "github.com/bex-co/bex/lego/types/v1alpha1"
	"os"
	"path/filepath"
	"testing"
)

func TestQA266ParkedSuccessfulDeployStatus(t *testing.T) {
	dir := os.Getenv("QA266_PROBE_DIR")
	raw, err := os.ReadFile(filepath.Join(dir, "sweep266-parked-snapshot_restart_saved.json"))
	if err != nil {
		t.Fatal(err)
	}
	var produced appv1alpha1.App
	if err := json.Unmarshal(raw, &produced); err != nil {
		t.Fatal(err)
	}
	if produced.Status.ActiveRevision != "rev-3" || produced.Status.ReleaseGeneration != 3 || produced.Status.Phase != appv1alpha1.PhaseHibernated {
		t.Fatal("operator producer must have served release3 and then parked")
	}
	open := Deploy{ID: "dep-qa266", AppID: "srv-qa266", Generation: 3, Status: DeployUpdateInProgress}
	for _, tc := range []struct {
		name       string
		phase      appv1alpha1.AppPhase
		active     string
		generation int64
		timedout   bool
		want       string
	}{
		{"awake completed control", appv1alpha1.PhaseRunning, "rev-3", 3, false, DeployLive},
		{"parked completed", appv1alpha1.PhaseHibernated, "rev-3", 3, false, DeployLive},
		{"parked completed after deadline", appv1alpha1.PhaseHibernated, "rev-3", 3, true, DeployLive},
		{"unserved parked control", appv1alpha1.PhaseHibernated, "rev-1", 3, false, ""},
		{"legacy unknown proof control", appv1alpha1.PhaseHibernated, "", 0, false, ""},
	} {
		t.Run(tc.name, func(t *testing.T) {
			app := produced.DeepCopy()
			app.Status.Phase = tc.phase
			app.Status.ActiveRevision = tc.active
			app.Status.ReleaseGeneration = tc.generation
			got := observedDeployStatus(open, app, tc.timedout)
			t.Logf("phase=%s release=%d active=%s timeout=%v got=%q want=%q", tc.phase, tc.generation, tc.active, tc.timedout, got, tc.want)
			if got != tc.want {
				t.Errorf("completed deploy status=%q, want %q", got, tc.want)
			}
		})
	}
}
