 var records []record
 for _,tc:=range []struct{name string;before,after map[string]string}{
 {"new-empty",nil,map[string]string{"QA_EMPTY":""}},
 {"existing-add-empty",map[string]string{"QA_EXISTING":"kept"},map[string]string{"QA_EMPTY":""}},
 {"existing-add-space",map[string]string{"QA_EXISTING":"kept"},map[string]string{"QA_EMPTY":" "}},

 {"existing-empty-noop",map[string]string{"QA_EMPTY":""},map[string]string{"QA_EMPTY":""}},
 {"existing-mixed-add",map[string]string{"QA_EXISTING":"kept"},map[string]string{"QA_EMPTY":"","QA_OTHER":"added"}},
 }{
 st:=newFakeStore();svc:=newService(st);ctx:=context.Background();old:=""
 if tc.before!=nil{if err:=svc.ApplyEnvGroup(ctx,"qa155-diagnostic",tc.before,nil);err!=nil{t.Fatal(err)};old,_,_,_=svc.findGroupByName(ctx,"qa155-diagnostic")}
