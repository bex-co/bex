const fs=require('node:fs'),http=require('node:http'),cp=require('node:child_process'),crypto=require('node:crypto');
const marker=JSON.parse(fs.readFileSync('qa433-marker.json')),fileNames=['qa433-app.cjs','qa433-transform.ts'];
const hash=b=>crypto.createHash('sha256').update(b).digest('hex');
function files(){return Object.fromEntries(fileNames.map(n=>[n,hash(fs.readFileSync(n))]))}
function version(command){return cp.execFileSync(command,['--version'],{encoding:'utf8',timeout:5000}).trim()}
function transform(label){const expected=Object.hasOwn(process.env,'BUN_VERSION'),r=cp.spawnSync('bun',['run','qa433-transform.ts'],{input:JSON.stringify({label,values:[2,3,5]}),encoding:'utf8',timeout:8000});
 if(r.error?.code==='ENOENT'){if(expected)throw Error('selected Bun missing');return {present:false,error:'ENOENT'}}
 if(r.error||r.status!==0)throw Error('Bun transform failed: '+r.stderr);
 const v=JSON.parse(r.stdout);if(!expected||!v.bun.startsWith('1.3.')||v.label!==label||v.sum!==10||v.weighted!==23||JSON.stringify(v.sorted)!=='[2,3,5]'||v.factor!==2||v.sha256!==hash(Buffer.from(label)))throw Error('actual Bun result mismatch');return {present:true,result:v}
}
const toolVersions={node:process.version,yarn:version('yarn'),yarnpkg:version('yarnpkg'),npm:version('npm')};
function snapshot(phase,label){const v={marker,phase,files:files(),tools:toolVersions,bun_env_present:Object.hasOwn(process.env,'BUN_VERSION'),bun_env:process.env.BUN_VERSION??null,bun:transform(label)};if(v.tools.yarn!=='1.22.22'||v.tools.yarnpkg!=='1.22.22')throw Error('Yarn aliases changed');return v}
if(process.argv[2]==='build'){const v=snapshot('build',marker+'|Crème brûlée & 李|build');fs.writeFileSync('qa433-built.json',JSON.stringify(v));console.log('QA433_BUILD '+JSON.stringify(v))}
else if(process.argv[2]==='serve'){const built=JSON.parse(fs.readFileSync('qa433-built.json'));if(JSON.stringify(built.files)!==JSON.stringify(files()))throw Error('build/runtime source changed');const boot=crypto.randomUUID(),runtime=snapshot('runtime',marker+'|Crème brûlée & 李|startup');console.log('QA433_RUNTIME '+JSON.stringify({boot,...runtime}));http.createServer((q,r)=>{try{const nonce=new URL(q.url,'http://local').searchParams.get('qa')??'root';if(nonce.length>100)throw Error('nonce too long');const label=marker+'|Crème brûlée & 李|'+nonce;const value={marker,boot,nonce,built,runtime:{...runtime,bun:transform(label)}};console.log('QA433_HTTP '+JSON.stringify({marker,boot,nonce,bun_present:value.runtime.bun.present}));r.writeHead(200,{'Content-Type':'application/json'});r.end(JSON.stringify(value))}catch(e){r.writeHead(500,{'Content-Type':'text/plain'});r.end(String(e))}}).listen(Number(process.env.PORT||3000),'0.0.0.0')}
else throw Error('bad mode');
