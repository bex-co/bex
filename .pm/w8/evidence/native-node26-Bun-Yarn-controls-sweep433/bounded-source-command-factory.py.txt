from pathlib import Path
import json,gzip,base64,shlex

def commands(P,marker):
 files={'qa433-app.cjs':(P/'sweep433-app.cjs').read_text(),'qa433-transform.ts':(P/'sweep433-transform.ts').read_text(),'qa433-marker.json':json.dumps(marker)}
 packed=base64.b64encode(gzip.compress(json.dumps(files,separators=(',',':'),ensure_ascii=False).encode(),mtime=0)).decode()
 js="const fs=require('node:fs'),z=require('node:zlib');for(const[n,s]of Object.entries(JSON.parse(z.gunzipSync(Buffer.from('"+packed+"','base64')))))fs.writeFileSync(n,s)"
 build='node -e '+shlex.quote(js)+'\nnode qa433-app.cjs build';assert len(build.encode())<4096
 return {'build':build,'start':'node qa433-app.cjs serve'}
