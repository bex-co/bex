  bundle:=filepath.Join(d,"env");must(t,os.WriteFile(bundle,[]byte("NODE_VERSION="+base64.StdEncoding.EncodeToString([]byte(tc.line))+"\n"),0600));df:=filepath.Join(d,"Dockerfile");must(t,os.WriteFile(df,[]byte(nativeDockerfile(o)),0600));choices,err:=ResolveNativeBuild("node",bundle,df,src);if err!=nil{t.Fatal(err)};resolved,err:=os.ReadFile(df);if err!=nil{t.Fatal(err)}

 })}
}
