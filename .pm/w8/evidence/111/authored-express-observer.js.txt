'use strict';
const fs=require('node:fs'), crypto=require('node:crypto'), path=require('node:path');
const express=require('./');
const reference=JSON.parse(fs.readFileSync('qa357-reference.json','utf8'));
const marker=JSON.parse(fs.readFileSync('qa357-marker.json','utf8'));
function sha(b){return crypto.createHash('sha256').update(b).digest('hex')}
function fileProof(){const out={};for(const [name,expected] of Object.entries(reference.files)){if(expected.absent){if(fs.existsSync(name))throw Error('unexpected file '+name);out[name]={absent:true}}else{const b=fs.readFileSync(name);out[name]={bytes:b.length,sha256:sha(b)};if(out[name].bytes!==expected.bytes||out[name].sha256!==expected.sha256)throw Error('source mismatch '+name)}}return out}
async function snapshot(){const app=express();let cacheOptions=[];app.engine('qa357',(name,options,callback)=>{cacheOptions.push(options.cache);callback(null,fs.readFileSync(name,'utf8'))});app.set('views',path.resolve('qa357-views'));app.set('view engine','qa357');const render=()=>new Promise((resolve,reject)=>app.render('index',(err,html)=>err?reject(err):resolve(html)));const rendered=[await render(),await render()];if(rendered.some(v=>v!==marker))throw Error('ordinary rendering failed');return {app,view:{marker,node:process.version,express_version:require('./package.json').version,node_env_present:Object.hasOwn(process.env,'NODE_ENV'),node_env:process.env.NODE_ENV??null,framework_env:app.get('env'),view_cache_enabled:app.enabled('view cache'),view_cache_entries:Object.keys(app.cache).length,render_cache_options:cacheOptions,rendered,files:fileProof()}}}
(async()=>{if(process.argv.includes('--build')){fs.mkdirSync('qa357-views',{recursive:true});fs.writeFileSync('qa357-views/index.qa357',marker);const {view}=await snapshot();fs.writeFileSync('qa357-built.json',JSON.stringify(view));console.log('QA357_BUILD '+JSON.stringify(view));return}const {app,view}=await snapshot();app.get('/',(req,res)=>res.json({...view,built:JSON.parse(fs.readFileSync('qa357-built.json','utf8'))}));app.listen(Number(process.env.PORT||10000),'0.0.0.0',()=>console.log('QA357_RUNTIME '+JSON.stringify(view)))})().catch(e=>{console.error(e);process.exit(1)});
