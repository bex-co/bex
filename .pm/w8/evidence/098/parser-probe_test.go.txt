package apps
import("encoding/json";"os";"testing"; validation "k8s.io/apimachinery/pkg/util/validation")
func TestQAKVReferenceCurrent175(t *testing.T){
 type input struct{Label string `json:"label"`;Document map[string]any `json:"document"`;Expected bool `json:"expected_valid_hypothesis"`}

  var target *parsedService;for i:=range st.services{if st.services[i].req.Name==targetName{target=&st.services[i]}};if target==nil{t.Fatal("target missing")};o.KVRefs=len(target.kvRefs)

 }

}
