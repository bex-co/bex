const fs=require('node:fs'),cp=require('node:child_process'),http=require('node:http'),crypto=require('node:crypto'),ref=require('./qa406-reference.json'),marker=require('./qa406-marker.json'),sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const files=Object.fromEntries(Object.keys(ref.files).map(p=>{const b=fs.readFileSync(p);return[p,{bytes:b.length,sha256:sha(b)}]}));if(JSON.stringify(files)!==JSON.stringify(ref.files))throw Error('source differs');
fs.writeFileSync('qa406.ppm',Buffer.from(ref.input.base64,'base64'));const input=fs.readFileSync('qa406.ppm');if(sha(input)!==ref.input.sha256)throw Error('input differs');
function tools(){return Object.fromEntries(['convert','identify'].map(t=>{const r=cp.spawnSync(t,['-version'],{encoding:'utf8',timeout:4000});return[t,{status:r.status,error:r.error?.code||null,stdout:r.stdout||'',stderr:r.stderr||''}]}))}
const base={marker,files,input:{bytes:input.length,sha256:sha(input)},node:process.version,program_sha256:sha(fs.readFileSync(__filename))};
if(process.argv[2]==='before'){const v={...base,tools:tools()};fs.writeFileSync('qa406-before.json',JSON.stringify(v));console.log('QA406_BEFORE '+JSON.stringify(v));process.exit(0)}
function image(b){const rgb=cp.execFileSync('convert',['png:-','-depth','8','rgb:-'],{input:b,timeout:4000}),dimensions=cp.execFileSync('identify',['-format','%m %w %h','png:-'],{input:b,timeout:4000}).toString();if(dimensions!=='PNG 2 1'||rgb.toString('hex')!==ref.expected_rgb_hex)throw Error('pixels differ');return{...base,tools:tools(),png:{bytes:b.length,sha256:sha(b),signature:b.subarray(0,8).toString('hex'),dimensions},pixels:{hex:rgb.toString('hex'),sha256:sha(rgb)}}}
if(process.argv[2]==='build'){const v={...image(fs.readFileSync('qa406.png')),before:require('./qa406-before.json')};fs.writeFileSync('qa406-build.json',JSON.stringify(v));console.log('QA406_AFTER '+JSON.stringify(v));process.exit(0)}
const built=require('./qa406-build.json');http.createServer((req,res)=>{const b=cp.execFileSync('convert',['qa406.ppm','-filter','point','-resize','2x1!','-strip','PNG24:-'],{timeout:4000}),runtime={...image(b),request_url:req.url},v={build:built,runtime};res.setHeader('Cache-Control','no-store');res.setHeader('Content-Type',req.url.startsWith('/png')?'image/png':'application/json');res.end(req.url.startsWith('/png')?b:JSON.stringify(v));if(req.url.includes('qa406='))console.log('QA406_REQUEST '+JSON.stringify(v))}).listen(Number(process.env.PORT||10000),process.env.QA406_LOCAL?'127.0.0.1':'0.0.0.0',function(){console.log('QA406_READY '+JSON.stringify({port:this.address().port,...base}))});
