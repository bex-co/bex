package apps
import("context";"encoding/json";"os";"strings";"testing";"github.com/bex-co/bex/lego/backend/internal/core")
func TestQAReferenceLocation195(t *testing.T){
 raw,err:=os.ReadFile(os.Getenv("BEX_QA_LOCATION195_INPUT"));if err!=nil{t.Fatal(err)}
 var input struct{Cases []struct{Label string `json:"label"`;Manifest string `json:"manifest"`;ExpectedEnvPointer string `json:"expectedEnvPointer"`;WrongPointer string `json:"wrongPointer"`;WrongField string `json:"wrongField"`;WrongFieldExists bool `json:"wrongFieldExists"`}}
 if err=json.Unmarshal(raw,&input);err!=nil{t.Fatal(err)}
 for _,tc:=range input.Cases{t.Run(tc.Label,func(t *testing.T){
  source,problems:=CompileBlueprintSource(tc.Manifest);if len(problems)>0{t.Fatalf("compiler problems: %+v",problems)}
  svc:=&Service{Base:&core.Base{Client:fakeClient(),Namespace:"default"}}
  got,err:=svc.ValidateBlueprint(context.Background(),"",tc.Manifest,"");if err!=nil||got.Valid||len(got.Errors)!=1{t.Fatalf("validation: %+v %v",got,err)}
  entry:=got.Errors[0];envloc:=source.Locations[tc.ExpectedEnvPointer];_,exists:=source.Locations[tc.WrongPointer]
  record:=map[string]any{"label":tc.Label,"actual":entry,"reported_pointer_exists":exists,"expected_env_pointer":tc.ExpectedEnvPointer,"expected_env_location":envloc}
  bytes,_:=json.Marshal(record);t.Log("QA195_LOCATION "+string(bytes))
  if exists!=tc.WrongFieldExists||envloc.Line==0||entry.Line==nil||*entry.Line==envloc.Line||entry.Path==nil||!strings.HasSuffix(*entry.Path,"services[0]."+tc.WrongField){t.Fatalf("current diagnostic bug not reproduced: %s",bytes)}
 })}
}
