package apps
import("context";"encoding/json";"os";"testing";"github.com/bex-co/bex/lego/backend/internal/core")
func TestQAPoolReferenceLocation194(t *testing.T){
 raw,err:=os.ReadFile(os.Getenv("BEX_QA_LOCATION194_INPUT"));if err!=nil{t.Fatal(err)}
 var input struct{Cases []struct{Label string `json:"label"`;Manifest string `json:"manifest"`;ExpectedEnvPointer string `json:"expectedEnvPointer"`;PhantomPointer string `json:"phantomPointer"`;ServicePointer string `json:"servicePointer"`}}
 if err=json.Unmarshal(raw,&input);err!=nil{t.Fatal(err)}
 for _,tc:=range input.Cases{t.Run(tc.Label,func(t *testing.T){
  source,problems:=CompileBlueprintSource(tc.Manifest);if len(problems)>0{t.Fatalf("compiler problems: %+v",problems)}
  svc:=&Service{Base:&core.Base{Client:fakeClient(),Namespace:"default"}}
  got,err:=svc.ValidateBlueprint(context.Background(),"",tc.Manifest,"");if err!=nil||got.Valid||len(got.Errors)!=1{t.Fatalf("validation: %+v %v",got,err)}
  entry:=got.Errors[0];envloc:=source.Locations[tc.ExpectedEnvPointer];parentloc:=source.Locations[tc.ServicePointer];_,exists:=source.Locations[tc.PhantomPointer]
  record:=map[string]any{"label":tc.Label,"actual":entry,"reported_pointer_exists":exists,"expected_env_pointer":tc.ExpectedEnvPointer,"expected_env_location":envloc,"service_location":parentloc}
  bytes,_:=json.Marshal(record);t.Log("QA194_LOCATION "+string(bytes))
  if exists||envloc.Line==0||entry.Line==nil||*entry.Line!=parentloc.Line||*entry.Line==envloc.Line{t.Fatalf("diagnostic did not reproduce current phantom-field/wrong-line behavior: %s",bytes)}
 })}
}
