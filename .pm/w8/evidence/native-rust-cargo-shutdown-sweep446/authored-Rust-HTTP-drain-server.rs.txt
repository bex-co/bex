use std::{fs,io::{Read,Write},net::{TcpListener,TcpStream},process::Command,sync::{Arc,atomic::{AtomicBool,AtomicUsize,Ordering}},thread,time::{Duration,Instant,SystemTime,UNIX_EPOCH}};
static STOP:AtomicBool=AtomicBool::new(false);static ACTIVE:AtomicUsize=AtomicUsize::new(0);
extern "C"{fn signal(sig:i32,handler:extern "C" fn(i32))->usize;}
extern "C" fn stop(_:i32){STOP.store(true,Ordering::SeqCst);}
fn sha(path:&str)->String{let v=Command::new("sha256sum").arg(path).output().unwrap();assert!(v.status.success());String::from_utf8(v.stdout).unwrap().split_whitespace().next().unwrap().to_string()}
fn proof()->String{assert_eq!(sha("LICENSE"),"20b064910b32bce1bc04595a5b20477a28c503995ef8528929a311d6cb7a3b09");assert_eq!(sha("CHANGELOG"),"9c65e366055edd9a0f6ab9c7b8a37fc92803cfe1d30ffa727ceed0c63939c2e5");let r=Command::new("rustc").arg("--version").output().unwrap();assert!(r.status.success());format!("\"marker\":{:?},\"rust\":{:?},\"program_sha256\":{:?}",include_str!("../marker.txt").trim(),String::from_utf8(r.stdout).unwrap().trim(),sha("qa446/src/main.rs"))}
fn event(phase:&str,boot:&str,token:&str){println!("QA446_EVENT {{\"marker\":{:?},\"boot\":{:?},\"token\":{:?},\"phase\":{:?},\"active\":{}}}",include_str!("../marker.txt").trim(),boot,token,phase,ACTIVE.load(Ordering::SeqCst));}
fn handle(mut stream:TcpStream,boot:Arc<String>,metadata:Arc<String>){stream.set_read_timeout(Some(Duration::from_secs(8))).unwrap();let mut bytes=Vec::new();let mut b=[0u8;1024];let header_end;
 loop{let n=stream.read(&mut b).unwrap();if n==0{return}bytes.extend_from_slice(&b[..n]);if let Some(i)=bytes.windows(4).position(|v|v==b"\r\n\r\n"){header_end=i+4;break}assert!(bytes.len()<16384);}
 let headers=String::from_utf8(bytes[..header_end].to_vec()).unwrap();let first=headers.lines().next().unwrap();let path=first.split_whitespace().nth(1).unwrap();let length=headers.lines().find_map(|s|s.to_ascii_lowercase().strip_prefix("content-length:").map(|v|v.trim().parse::<usize>().unwrap())).unwrap_or(0);assert!(length<=4096);while bytes.len()-header_end<length{let n=stream.read(&mut b).unwrap();if n==0{return}bytes.extend_from_slice(&b[..n]);}
 let token=path.strip_prefix("/hold/").unwrap_or("");assert!(token.bytes().all(|x|x.is_ascii_alphanumeric()||x==b'-'||x==b'_'));let body_hex=bytes[header_end..header_end+length].iter().map(|b|format!("{:02x}",b)).collect::<String>();
 if !token.is_empty(){ACTIVE.fetch_add(1,Ordering::SeqCst);event("hold-begin",&boot,token);thread::sleep(Duration::from_secs(40));event("hold-end",&boot,token);}
 let response=format!("{{{},\"boot\":{:?},\"token\":{:?},\"body_hex\":{:?}}}",metadata,boot.as_str(),token,body_hex);write!(stream,"HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{}",response.len(),response).unwrap();stream.flush().unwrap();
 if !token.is_empty(){ACTIVE.fetch_sub(1,Ordering::SeqCst);}
}
fn main(){let p=proof();if std::env::args().any(|v|v=="--build-proof"){fs::write("qa446/build-proof.json",format!("{{{}}}",p)).unwrap();println!("QA446_BUILD {{{}}}",p);return}unsafe{signal(15,stop);}let boot=Arc::new(format!("{}-{}",std::process::id(),SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_nanos()));let metadata=Arc::new(p);let port=std::env::var("PORT").unwrap_or("3000".into());let listener=TcpListener::bind(format!("0.0.0.0:{}",port)).unwrap();listener.set_nonblocking(true).unwrap();event("ready",&boot,"");while !STOP.load(Ordering::SeqCst){match listener.accept(){Ok((stream,_))=>{let boot=boot.clone();let metadata=metadata.clone();thread::spawn(move||handle(stream,boot,metadata));},Err(e) if e.kind()==std::io::ErrorKind::WouldBlock=>thread::sleep(Duration::from_millis(10)),Err(e)=>panic!("{}",e)}}event("sigterm",&boot,"");let end=Instant::now();while ACTIVE.load(Ordering::SeqCst)>0&&end.elapsed()<Duration::from_secs(44){thread::sleep(Duration::from_millis(10));}event("closed",&boot,"");}
