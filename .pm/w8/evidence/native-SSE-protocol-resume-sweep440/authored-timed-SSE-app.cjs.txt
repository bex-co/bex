const fs=require('node:fs'),http=require('node:http'),crypto=require('node:crypto');
const marker=JSON.parse(fs.readFileSync('qa440-marker.json')),ref=JSON.parse(fs.readFileSync('qa440-reference.json')),hash=b=>crypto.createHash('sha256').update(b).digest('hex');
const source=()=>Object.fromEntries(Object.keys(ref.files).map(n=>{const b=fs.readFileSync(n);return[n,{bytes:b.length,sha256:hash(b)}]}));
function proof(){const files=source();if(JSON.stringify(files)!==JSON.stringify(ref.files))throw Error('source mismatch');return{marker,files,node:process.version,program_sha256:hash(fs.readFileSync(__filename))}}
if(process.argv[2]==='build'){const v=proof();fs.writeFileSync('qa440-built.json',JSON.stringify(v));console.log('QA440_BUILD '+JSON.stringify(v))}
else if(process.argv[2]==='serve'){const built=JSON.parse(fs.readFileSync('qa440-built.json')),boot=crypto.randomUUID();http.createServer((q,r)=>{const u=new URL(q.url,'http://local');if(u.pathname!='/events'){r.writeHead(200,{'Content-Type':'application/json'});r.end(JSON.stringify({...proof(),built,boot}));return}const nonce=u.searchParams.get('run')||'default',last=Number(q.headers['last-event-id']||0),count=Number(u.searchParams.get('count')||3);if(nonce.length>80||!Number.isInteger(last)||last<0||last>10||count!==3){r.writeHead(400);r.end('invalid own fixture input');return}r.writeHead(200,{'Content-Type':'text/event-stream; charset=utf-8','Cache-Control':'no-cache','X-Accel-Buffering':'no'});r.flushHeaders();let sent=0,timer;console.log('QA440_STREAM '+JSON.stringify({marker,boot,nonce,last,phase:'begin'}));const emit=()=>{const seq=last+ ++sent;const payload={marker,boot,nonce,seq,text:'Crème brûlée & 李',files:source()};const json=JSON.stringify(payload),at=json.indexOf(',"text"');const frame=Buffer.from(': heartbeat\r\n'+(sent===1?'retry: 500\r\n':'')+'id: '+seq+'\r\nevent: tick\r\ndata: '+json.slice(0,at+1)+'\r\ndata: '+json.slice(at+1)+'\r\n\r\n');const chinese=Buffer.from('李'),pos=frame.indexOf(chinese);const cut=pos>=0?pos+1:Math.floor(frame.length/2);r.write(frame.subarray(0,cut));setTimeout(()=>{if(r.destroyed)return;r.write(frame.subarray(cut));console.log('QA440_EVENT '+JSON.stringify({marker,boot,nonce,seq}));if(sent===count){r.end();console.log('QA440_STREAM '+JSON.stringify({marker,boot,nonce,last,phase:'end',sent}))}else timer=setTimeout(emit,750)},15)};timer=setTimeout(emit,200);r.on('close',()=>clearTimeout(timer))}).listen(Number(process.env.PORT||3000),'0.0.0.0',()=>console.log('QA440_READY '+JSON.stringify({marker,boot,node:process.version})))}else throw Error('mode');
