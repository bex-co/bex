def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def save():
    c['last_updated'] = now()
    temporary = CP.with_name('.'+CP.name+'.qa308.tmp')
    temporary.write_text(json.dumps(c, indent=2, ensure_ascii=False))
    temporary.chmod(0o600)
    temporary.replace(CP)

def cli(args, bound=45):
    cfg = P / 'render-config/cli.yaml'
    try:
        if client == 'render':
            assert not cfg.exists()
            shutil.copyfile(P / 'config/cli.yaml', cfg)
            cfg.chmod(0o600)
        runner = P / ('run-render.py' if client == 'render' else 'run.py')
        r = json.loads(subprocess.check_output(['python3', str(runner), *([] if client=='render' else ['--release']), '--bound=' + str(bound), *args], text=True, timeout=bound + 15))
        r['native_exit'] = None if r['exit'] == 124 else r['exit']
        return r
    finally:
        if client == 'render':
            cfg.unlink(missing_ok=True)

def wire(path):

def inventory():
    r = cli(['services', '--confirm', '-o', 'json'])
    assert r['native_exit'] == 0
    rows = [v.get('service', v) for v in json.loads(r['stdout']) or []]
    assert BASE <= {v['id'] for v in rows}
    return r, rows

def history(x):
    r = cli(['deploys', 'list', x['id'], '--confirm', '-o', 'json'])
    assert r['native_exit'] == 0
    rows = json.loads(r['stdout'])
    x.setdefault('history_receipts', []).append({'at': now(), 'result': r})
    x['deploy_children'] = rows
    for row in rows:
        if row['id'] not in x['recorded_child_ids']:
            x['recorded_child_ids'].append(row['id'])
    save()
    return rows

def http(x, path, method='GET'):
    body = P / 'sweep308-body'
    headers = P / 'sweep308-headers'
    assert not body.exists() and not headers.exists()
    url = x['create_response']['serviceDetails']['url'] + path
    start = time.monotonic()
    try:
        args = ['curl', '--disable', '-sS', '--max-time', '12', '--path-as-is', '-H', 'Accept-Encoding: identity', '-D', str(headers), '-o', str(body), '-w', '%{http_code}']
        args += ['--head'] if method == 'HEAD' else []
        r = subprocess.run([*args, url], text=True, capture_output=True, timeout=17)
        data = body.read_bytes() if body.exists() and method != 'HEAD' else b''
        hs = {}
        if headers.exists():
            for line in headers.read_text().splitlines():
                k, sep, v = line.partition(':')
                if sep and k.lower() in ('content-type', 'content-length', 'etag', 'last-modified', 'location', 'cache-control', 'retry-after', 'x-qa308'):
                    hs[k.lower()] = v.strip()
        return {'at': now(), 'request': {'method': method, 'url': url, 'headers': {'Accept-Encoding': 'identity'}}, 'HTTP': r.stdout, 'curl_exit': r.returncode, 'elapsed': round(time.monotonic()-start, 3), 'response_headers': hs, 'body_utf8': data.decode('utf8', errors='replace'), 'bytes': len(data), 'stderr': r.stderr}
    finally:
        body.unlink(missing_ok=True)
        headers.unlink(missing_ok=True)
