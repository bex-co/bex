def http(x, path, method='GET'):
    body = P / 'sweep308-body'
    headers = P / 'sweep308-headers'
    assert not body.exists() and not headers.exists()
    url = x['create_response']['serviceDetails']['url'] + path
    start = time.monotonic()
    try:
        args = ['curl', '--disable', '-sS', '--max-time', '12', '--path-as-is', '-H', 'Accept-Encoding: identity', '-D', str(headers), '-o', str(body), '-w', '%{http_code}']
        args += ['--head'] if method == 'HEAD' else []
        r = subprocess.run([*args, url], text=True, capture_output=True, timeout=17)
        data = body.read_bytes() if body.exists() and method != 'HEAD' else b''
        hs = {}
        if headers.exists():
            for line in headers.read_text().splitlines():
                k, sep, v = line.partition(':')
                if sep and k.lower() in ('content-type', 'content-length', 'etag', 'last-modified', 'location', 'cache-control', 'retry-after', 'x-qa308'):
                    hs[k.lower()] = v.strip()
        return {'at': now(), 'request': {'method': method, 'url': url, 'headers': {'Accept-Encoding': 'identity'}}, 'HTTP': r.stdout, 'curl_exit': r.returncode, 'elapsed': round(time.monotonic()-start, 3), 'response_headers': hs, 'body_utf8': data.decode('utf8', errors='replace'), 'bytes': len(data), 'stderr': r.stderr}
    finally:
        body.unlink(missing_ok=True)
        headers.unlink(missing_ok=True)
