188: // pageCursors computes the Render paging envelope fields shared by REST,
189: // GraphQL, and MCP. Callers feed nextStartTime/nextEndTime back as
190: // startTime/endTime to fetch the next page.
191: //
192: // Boundary rule (start and end are inclusive on the pod-log path; Loki's
193: // query_range end is exclusive — both stay correct with the adjustments
194: // below):
195: //
196: //   - backward (default): nextStartTime = query start, nextEndTime = one
197: //     nanosecond before the page's oldest timestamp — so an inclusive end
198: //     does not re-include the boundary group. capToLimit keeps a shared-
199: //     timestamp group whole first, so every sibling at that instant is
200: //     already on this page.
201: //   - forward: nextStartTime = one nanosecond past the page's newest
202: //     timestamp, nextEndTime = query end — so an inclusive start does not
203: //     repeat the newest line (same whole-group guarantee from capToLimit).
204: func pageCursors(entries []LogEntry, limit int64, since, end time.Time, direction string) (hasMore bool, nextStart, nextEnd string) {
205: 	since, end = resolveCursorWindow(since, end)
206: 	hasMore = limit > 0 && int64(len(entries)) >= limit
207: 	if len(entries) == 0 {
208: 		// Second precision matches Render's empty-page cursors and keeps two
209: 		// identical empty reads byte-identical within the same second.
210: 		return hasMore, since.UTC().Format(time.RFC3339), end.UTC().Format(time.RFC3339)
211: 	}
212: 	oldest, newest := entries[0].Timestamp, entries[len(entries)-1].Timestamp
213: 	if direction == DirectionForward {
214: 		return hasMore, advancePast(newest), end.UTC().Format(time.RFC3339Nano)
215: 	}
216: 	return hasMore, since.UTC().Format(time.RFC3339Nano), retreatBefore(oldest)
217: }
