const fs=require('node:fs'),http=require('node:http'),crypto=require('node:crypto');
const marker=JSON.parse(fs.readFileSync('qa441-marker.json')),ref=JSON.parse(fs.readFileSync('qa441-reference.json')),sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const source=()=>Object.fromEntries(Object.keys(ref.files).map(n=>{const b=fs.readFileSync(n);return[n,{bytes:b.length,sha256:sha(b)}]}));
function proof(){const files=source();if(JSON.stringify(files)!==JSON.stringify(ref.files))throw Error('source mismatch');return{marker,node:process.version,files,program_sha256:sha(fs.readFileSync(__filename))}}
if(process.argv[2]==='build'){const v=proof();fs.writeFileSync('qa441-built.json',JSON.stringify(v));console.log('QA441_BUILD '+JSON.stringify(v))}
else if(process.argv[2]==='serve'){const boot=crypto.randomUUID(),built=JSON.parse(fs.readFileSync('qa441-built.json'));const server=http.createServer((q,r)=>{if(!q.url.startsWith('/upload/')){r.writeHead(200,{'Content-Type':'application/json'});r.end(JSON.stringify({...proof(),built,boot,requestTimeout:server.requestTimeout}));return}const nonce=q.url.slice(8);if(!/^[a-zA-Z0-9_-]{1,80}$/.test(nonce)||q.method!=='PUT'){r.writeHead(400);r.end('invalid own upload fixture');return}let bytes=0,last=0,ended=false;const hash=crypto.createHash('sha256'),at=Date.now();const event=(phase,extra={})=>console.log('QA441_EVENT '+JSON.stringify({marker,boot,nonce,phase,bytes,elapsed_ms:Date.now()-at,...extra}));event('begin',{declared:q.headers['content-length'],method:q.method});q.on('data',b=>{bytes+=b.length;hash.update(b);if(bytes-last>=65536){last=bytes;event('progress')}});q.on('end',()=>{ended=true;const digest=hash.digest('hex');event('complete',{sha256:digest});r.writeHead(200,{'Content-Type':'application/json'});r.end(JSON.stringify({marker,boot,nonce,bytes,sha256:digest,elapsed_ms:Date.now()-at,files:source()}))});q.on('aborted',()=>event('aborted',{partial_sha256:hash.copy().digest('hex')}));q.on('error',e=>event('request-error',{code:e.code}));q.on('close',()=>{if(!ended)event('closed-before-end')})});server.requestTimeout=180000;server.headersTimeout=60000;server.listen(Number(process.env.PORT||3000),'0.0.0.0',()=>console.log('QA441_READY '+JSON.stringify({marker,boot,node:process.version,requestTimeout:server.requestTimeout})))}else throw Error('mode');
