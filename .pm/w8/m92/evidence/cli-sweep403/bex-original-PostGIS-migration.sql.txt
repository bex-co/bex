BEGIN; SET LOCAL statement_timeout='20s'; CREATE EXTENSION IF NOT EXISTS postgis; WITH p AS (SELECT ST_SetSRID(ST_MakePoint(0,0),4326) a,ST_SetSRID(ST_MakePoint(1,0),4326) b) SELECT jsonb_build_object('srid',ST_SRID(a),'wkt',ST_AsText(a),'distance_m',round(ST_Distance(a::geography,b::geography)::numeric,3),'extension',(SELECT extversion FROM pg_extension WHERE extname='postgis'))::text FROM p; ROLLBACK;
