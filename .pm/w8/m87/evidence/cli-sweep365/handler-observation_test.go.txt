 for _, keyBytes := range []int{4,32} { t.Run(map[int]string{4:"protocol30",32:"protocol32"}[keyBytes],func(t *testing.T){
  ln,err:=net.Listen("tcp","127.0.0.1:0");if err!=nil {t.Fatal(err)};defer ln.Close()
  logger:=qa365Logger{make(chan string,8)};done:=make(chan struct{})
  go func(){defer close(done);conn,err:=ln.Accept();if err!=nil{return};handleConn(conn,newRouter("db.example.invalid"),sniproxy.NewByteMeter(prometheus.NewRegistry(),"qa365","postgres"),nil,sniproxy.NewLimiter(0,0),0,0,logger)}()
  conn,err:=net.DialTimeout("tcp",ln.Addr().String(),time.Second);if err!=nil{t.Fatal(err)};defer conn.Close()
  packet:=make([]byte,12+keyBytes);binary.BigEndian.PutUint32(packet[0:4],uint32(len(packet)));binary.BigEndian.PutUint32(packet[4:8],80877102);binary.BigEndian.PutUint32(packet[8:12],12345);for i:=12;i<len(packet);i++{packet[i]=byte(i)}
  if _,err=conn.Write(packet);err!=nil{t.Fatal(err)}
  select{case <-done:case <-time.After(2*time.Second):t.Fatal("handler did not finish")}
  select{case msg:=<-logger.messages:if msg!="unknown protocol header, closing"{t.Fatalf("unexpected result %q",msg)};t.Logf("valid CancelRequest length=%d, code=80877102: %s before route lookup",len(packet),msg);default:t.Fatal("no handler observation")}
 }) }
}
