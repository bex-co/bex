package build

import (
 "encoding/base64"
 "encoding/json"
 "os"
 "path/filepath"
 "strings"
 "testing"
)

func TestQA306BunResolverObservation(t *testing.T) {
 out:=os.Getenv("QA306_OUTPUT")
 if out=="" {t.Fatal("private output directory required")}
 for _,runtime:=range []string{"node","python","ruby","go","rust","elixir"} {
  t.Run(runtime,func(t *testing.T){
   dir:=t.TempDir()
   source:=filepath.Join(dir,"source")
   if err:=os.Mkdir(source,0700);err!=nil{t.Fatal(err)}
   bundle:=filepath.Join(dir,"env")
   if err:=os.WriteFile(bundle,[]byte("BUN_VERSION="+base64.StdEncoding.EncodeToString([]byte("1.3.4"))+"\n"),0600);err!=nil{t.Fatal(err)}
   file:=filepath.Join(dir,"Dockerfile")
   before:=nativeDockerfile(Options{Runtime:runtime,BuildCommand:"printf '%s\\n' QA306_SOURCE",StartCommand:"printf '%s\\n' QA306_START"})
   if err:=os.WriteFile(file,[]byte(before),0600);err!=nil{t.Fatal(err)}
   choice,err:=ResolveNativeBuild(runtime,bundle,file,source)
   if err!=nil{t.Fatal(err)}
   after,err:=os.ReadFile(file);if err!=nil{t.Fatal(err)}
   env,err:=readEnvBundle(bundle);if err!=nil{t.Fatal(err)}
   record:=struct{Runtime string `json:"runtime"`;Requested string `json:"requested_bun"`;Choice NativeChoice `json:"primary_choice"`;Unchanged bool `json:"dockerfile_unchanged"`;BunText bool `json:"has_bun_setup_text"`}{runtime,env["BUN_VERSION"],choice,before==string(after),strings.Contains(strings.ToLower(string(after)),"bun")}
   data,err:=json.MarshalIndent(record,"","  ");if err!=nil{t.Fatal(err)}
   for name,content:=range map[string][]byte{runtime+".json":data,runtime+".Dockerfile":after}{if err:=os.WriteFile(filepath.Join(out,name),content,0600);err!=nil{t.Fatal(err)}}
   t.Logf("runtime=%s BUN_VERSION_decoded=%s primary=%s line=%s resolver_changed_Dockerfile=%v Bun_setup_text=%v",runtime,record.Requested,choice.Label,choice.Line,!record.Unchanged,record.BunText)
   // Observation only: this is not a regression or a claimed fix.
   if record.Requested!="1.3.4"{t.Fatal("fixture env did not reach actual resolver")}
  })
 }
}
